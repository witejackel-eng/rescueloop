   797→// ─── Transactional outbox ────────────────────────────────────
   798→// Ensures job dispatch is durable. When application state requires a job:
   799→// 1. Write the domain mutation + OutboxEvent in the same DB transaction
   800→// 2. Commit
   801→// 3. Dispatcher reads pending events and dispatches to Inngest
   802→// 4. Marks dispatched (or failed → dead-letter after retries)
   803→
   804→enum OutboxState {
   805→  pending
   806→  dispatching
   807→  dispatched
   808→  failed
   809→  dead_letter
   810→}
   811→
   812→model OutboxEvent {
   813→  id              String      @id @default(cuid())
   814→  organizationId  String
   815→  eventType       String // e.g. "whop/webhook.received", "deliver/intervention"
   816→  payloadJson     Json
   817→  state           OutboxState @default(pending)
   818→  idempotencyKey  String      @unique
   819→  attemptCount    Int         @default(0)
   820→  maxAttempts     Int         @default(5)
   821→  lastError       String?
   822→  dispatchedAt    DateTime?   @db.Timestamptz
   823→  nextAttemptAt   DateTime?   @db.Timestamptz
   824→  externalEventId String? // Inngest event ID after successful dispatch
   825→  claimedBy       String? // Worker identifier that holds the lease
   826→  claimedAt       DateTime?   @db.Timestamptz // When the lease was acquired
   827→  leaseExpiresAt  DateTime?   @db.Timestamptz // When the lease expires
   828→  createdAt       DateTime    @default(now()) @db.Timestamptz
   829→  updatedAt       DateTime    @updatedAt @db.Timestamptz
   830→
   831→  @@index([organizationId, state])
   832→  @@index([state, nextAttemptAt])
   833→  @@index([state, claimedBy, leaseExpiresAt])
   834→  @@map("outbox_events")
   835→}
