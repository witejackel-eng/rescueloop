   800→// ─── Transactional outbox ────────────────────────────────────
   801→// Ensures job dispatch is durable. When application state requires a job:
   802→// 1. Write the domain mutation + OutboxEvent in the same DB transaction
   803→// 2. Commit
   804→// 3. Dispatcher reads pending events and dispatches to Inngest
   805→// 4. Marks dispatched (or failed → dead-letter after retries)
   806→

   808→  pending
   809→  dispatching
   810→  dispatched
   811→  failed
   812→  dead_letter
   813→}

   822→  attemptCount    Int         @default(0)
   823→  maxAttempts     Int         @default(5)
   824→  lastError       String?
   825→  dispatchedAt    DateTime?   @db.Timestamptz
   826→  nextAttemptAt   DateTime?   @db.Timestamptz
   827→  externalEventId String? // Inngest event ID after successful dispatch
   828→  claimedBy       String? // Worker identifier that holds the lease

   830→  leaseExpiresAt  DateTime?   @db.Timestamptz // When the lease expires

   833→
   834→  @@index([organizationId, state])
