   797→   797→// ─── Transactional outbox ────────────────────────────────────
   798→   798→// Ensures job dispatch is durable. When application state requires a job:
   799→   799→// 1. Write the domain mutation + OutboxEvent in the same DB transaction
   800→   800→// 2. Commit
   801→   801→// 3. Dispatcher reads pending events and dispatches to Inngest
   802→   802→// 4. Marks dispatched (or failed → dead-letter after retries)
   803→   803→
   804→   804→enum OutboxState {
   805→   805→  pending
   806→   806→  dispatching
   807→   807→  dispatched
   808→   808→  failed
   809→   809→  dead_letter
   810→   810→}

   818→   818→  idempotencyKey  String      @unique
   819→   819→  attemptCount    Int         @default(0)
   820→   820→  maxAttempts     Int         @default(5)
   821→   821→  lastError       String?

   824→   824→  externalEventId String? // Inngest event ID after successful dispatch
   825→   825→  claimedBy       String? // Worker identifier that holds the lease

   827→   827→  leaseExpiresAt  DateTime?   @db.Timestamptz // When the lease expires
