   853→// ─── Transactional outbox ────────────────────────────────────
   854→// Ensures job dispatch is durable. When application state requires a job:
   855→// 1. Write the domain mutation + OutboxEvent in the same DB transaction
   856→// 2. Commit
   857→// 3. Dispatcher reads pending events and dispatches to Inngest
   858→// 4. Marks dispatched (or failed → dead-letter after retries)
   859→

   861→  pending
   862→  dispatching
   863→  dispatched
   864→  failed
   865→  dead_letter
   866→}

   875→  attemptCount    Int         @default(0)

   877→  lastError       String?
   878→  dispatchedAt    DateTime?   @db.Timestamptz
   879→  nextAttemptAt   DateTime?   @db.Timestamptz
   880→  externalEventId String? // Inngest event ID after successful dispatch
   881→  claimedBy       String? // Worker identifier that holds the lease
   882→  claimedAt       DateTime?   @db.Timestamptz // When the lease was acquired
   883→  leaseExpiresAt  DateTime?   @db.Timestamptz // When the lease expires
   884→  createdAt       DateTime    @default(now()) @db.Timestamptz
   885→  updatedAt       DateTime    @updatedAt @db.Timestamptz
   886→
   887→  @@index([organizationId, state])
