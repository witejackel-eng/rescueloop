     6→// demo routes (/overview, /rescue-queue, etc.). Every nav href includes
     7→// the companyId so creators stay inside their company scope.

    10→// visually distinct from the demo workspace:
    11→//   - Wider desktop nav rail (76px) with section dividers
    12→//   - Environment badge (FIXTURE / CONNECTED / NOT CONFIGURED)
    13→//   - Installation-state dot next to the company name
    14→//   - "Last sync" timestamp with relative time
    15→//   - Emergency pause button (calls /api/companies/[id]/settings/pause)
    16→//   - User menu placeholder (no auth wired here — server-side handles it)
    17→//
    18→// All numbers in font-mono tabular-nums.

   712→function formatRelativeShort(iso: string): string {
   713→  const d = new Date(iso);
   714→  const diffMs = Date.now() - d.getTime();
   715→  const mins = Math.floor(diffMs / 60000);
   716→  if (mins < 1) return "just now";
   717→  if (mins < 60) return `${mins}m ago`;
   718→  const hours = Math.floor(mins / 60);
   719→  if (hours < 24) return `${hours}h ago`;
   720→  const days = Math.floor(hours / 24);
   721→  if (days < 30) return `${days}d ago`;
   722→  return d.toLocaleDateString("en-US", { month: "short", day: "numeric" });
   723→}
