   107→/**
   108→ * Classify a Prisma / DB error into a structured failure type.
   109→ * Unique constraint violations are "duplicate" — everything else is a real error.
   110→ *
   111→ * Phase 10: Also detects student_not_found, course_not_found, and validation_error
   112→ * based on the error message content, so callers never treat these as duplicates.
   113→ */
   114→export function classifyCreateError(error: unknown, context: string): ClassifiedFailure {
   115→  const msg = error instanceof Error ? error.message : String(error ?? "unknown");
   116→
   117→  // Foreign key constraint violation — Prisma P2003
   118→  if (
   119→    msg.includes("Foreign key constraint failed") ||
   120→    msg.includes("P2003")
   121→  ) {
   122→    if (context.includes("student") || msg.includes("studentId") || msg.includes("Student")) {
   123→      return { class: "student_not_found", externalId: context, message: msg };
   124→    }
   125→    if (context.includes("course") || msg.includes("courseId") || msg.includes("Course")) {
   126→      return { class: "course_not_found", externalId: context, message: msg };
   127→    }
   128→    return { class: "db_error", externalId: context, message: msg };
   129→  }
   130→
   131→  // Validation error — Prisma P2025 (record not found) or data validation
   132→  if (
   133→    msg.includes("required") && msg.includes("Argument") ||
   134→    msg.includes("P2025") ||
   135→    msg.includes("validation")
   136→  ) {
   137→    return { class: "validation_error", externalId: context, message: msg };
   138→  }
   139→
   140→  // Prisma unique constraint violation codes
   141→  if (
   142→    msg.includes("Unique constraint failed") ||
   143→    msg.includes("Unique constraint violation") ||
   144→    msg.includes("P2002") // Prisma error code for unique constraint
   145→  ) {
   146→    // Determine which constraint based on context or message content
   147→    if (context.includes("payloadHash") || msg.includes("payloadHash")) {
   148→      return { class: "duplicate_payload_hash", externalId: context, message: msg };
   149→    }
   150→    return { class: "duplicate_unique_constraint", externalId: context, message: msg };
   151→  }
   152→
   153→  return { class: "db_error", externalId: context, message: msg };
   154→}

   156→// ─── Payload hash for dedup (Phase 10) ──────────────────────
   157→
   158→/**
   159→ * Simple deterministic hash for progress event dedup.
   160→ * Produces a hex string from the interaction's identifying fields.
   161→ * In production, replace with crypto.subtle.digest("SHA-256", ...).
   162→ */
   163→export function computePayloadHash(fields: {
   164→  userId: string;
   165→  courseId: string;
   166→  lessonId: string;
   167→  action: string;
   168→  sourceTimestamp: string;
   169→}): string {
   170→  // Deterministic concatenation for hashing
   171→  const raw = `${fields.userId}|${fields.courseId}|${fields.lessonId}|${fields.action}|${fields.sourceTimestamp}`;
   172→  // Simple FNV-1a hash for non-crypto dedup (good enough for idempotency keys)
   173→  let hash = 0x811c9dc5;
   174→  for (let i = 0; i < raw.length; i++) {
   175→    hash ^= raw.charCodeAt(i);
   176→    hash = Math.imul(hash, 0x01000193);
   177→  }
   178→  // Convert to unsigned 32-bit hex
   179→  return (hash >>> 0).toString(16).padStart(8, "0");
   180→}

   182→// ─── Bounded concurrency helper ──────────────────────────────
   183→
   184→/** Run items with at most `concurrency` operations in flight. */
   185→async function boundedMap<T, R>(
   186→  items: T[],
   187→  fn: (item: T) => Promise<R>,
   188→  concurrency: number = MAX_CONCURRENT_UPDATES,
   189→): Promise<R[]> {
   190→  const results: R[] = [];
   191→  const executing: Promise<void>[] = [];
   192→
   193→  for (const item of items) {
   194→    const p = fn(item).then((r) => { results.push(r); });
   195→    executing.push(p);
   196→
   197→    if (executing.length >= concurrency) {
   198→      await Promise.race(executing);
   199→      // Remove settled promises
   200→      for (let i = executing.length - 1; i >= 0; i--) {
   201→        const settled = await Promise.race([
   202→          executing[i].then(() => true, () => true),
   203→          Promise.resolve(false),
   204→        ]);
   205→        if (settled) executing.splice(i, 1);
   206→      }
   207→    }
   208→  }
   209→
   210→  await Promise.allSettled(executing);
   211→  return results;
   212→}

   216→export async function syncCourses(params: SyncExecutionParams): Promise<SyncResult> {
   217→  const result: SyncResult = {
   218→    resource: "courses",
   219→    recordsRead: 0,
   220→    recordsCreated: 0,
   221→    recordsUpdated: 0,
   222→    recordsSkipped: 0,
   223→    errors: [],
   224→    warnings: [],
   225→    cursor: null,
   226→  };
   227→
   228→  // Resume from checkpoint if no explicit cursor
   229→  let cursor = params.cursor ?? null;
   230→  if (!cursor) {
   231→    const checkpoint = await getLatestCheckpoint(params.organizationId, "courses");
   232→    if (checkpoint?.cursor) {
   233→      cursor = checkpoint.cursor;
   234→    }
   235→  }
   236→
   237→  let pagesProcessed = 0;
   238→
   239→  do {
   240→    const page = await params.providers.courses.list({
   241→      companyId: params.companyId,
   242→      cursor,
   243→      pageSize: PAGE_SIZE,
   244→    });
   245→
   246→    result.recordsRead += page.items.length;
   247→    pagesProcessed++;
   248→
   249→    // 1. Validate the full page
   250→    const validItems = page.items.filter((course) => {
   251→      if (!course.id) {
   252→        result.errors.push(`Course missing id — skipped`);
   253→        result.recordsSkipped++;
   254→        return false;
   255→      }
   256→      return true;
   257→    });
   258→
   259→    if (validItems.length === 0) {
   260→      cursor = page.nextCursor;
   261→      continue;
   262→    }
   263→
   264→    // 2. Load existing external IDs in one bounded query
   265→    const externalIds = validItems.map((c) => c.id);
   266→    const existing = await db.course.findMany({
   267→      where: {
   268→        organizationId: params.organizationId,
   269→        externalCourseId: { in: externalIds },
   270→      },
   271→      select: { id: true, externalCourseId: true },
   272→    });
   273→    const existingMap = new Map(existing.map((e) => [e.externalCourseId, e.id]));
   274→
   275→    // 3. Split into creates and updates
   276→    const toCreate = validItems.filter((c) => !existingMap.has(c.id));
   277→    const toUpdate = validItems.filter((c) => existingMap.has(c.id));
   278→
   279→    // 4. Batch creates with skipDuplicates
   280→    if (toCreate.length > 0) {
   281→      try {
   282→        await db.course.createMany({
   283→          data: toCreate.map((course) => ({
   284→            organizationId: params.organizationId,
   285→            externalCourseId: course.id,
   286→            externalExperienceId: course.experienceId,
   287→            name: course.title ?? "Untitled Course",
   288→            lessonCount: course.lessonCount,
   289→          })),
   290→          skipDuplicates: true,
   291→        });
   292→        result.recordsCreated += toCreate.length;
   293→      } catch (error) {
   294→        result.errors.push(`Course batch create: ${error instanceof Error ? error.message : "unknown"}`);
   295→        result.recordsSkipped += toCreate.length;
   296→      }
   297→    }
   298→
   299→    // 5. Batch updates with bounded concurrency
   300→    if (toUpdate.length > 0) {
   301→      const updateResults = await boundedMap(toUpdate, async (course) => {
   302→        const existingId = existingMap.get(course.id)!;
   303→        try {
   304→          await db.course.update({
   305→            where: { id: existingId },
   306→            data: {
   307→              name: course.title ?? undefined,
   308→              lessonCount: course.lessonCount || undefined,
   309→              externalExperienceId: course.experienceId || undefined,
   310→            },
   311→          });
   312→          return true;
   313→        } catch (error) {
   314→          result.errors.push(`Course ${course.id}: ${error instanceof Error ? error.message : "unknown"}`);
   315→          return false;
   316→        }
   317→      });
   318→      result.recordsUpdated += updateResults.filter(Boolean).length;
   319→      result.recordsSkipped += updateResults.filter((r) => !r).length;
   320→    }
   321→
   322→    // 6. Commit page — persist checkpoint
   323→    cursor = page.nextCursor;
   324→    if (params.syncExecutionId) {
   325→      await persistCheckpoint(
   326→        params.organizationId,
   327→        "courses",
   328→        cursor,
   329→        null,
   330→        params.syncExecutionId,
   331→        pagesProcessed,
   332→      );
   333→    }
   334→  } while (cursor);
   335→
   336→  result.cursor = cursor;
   337→
   338→  // 7. Emit usage and audit data
   339→  await recordAuditEvent({
   340→    organizationId: params.organizationId,
   341→    action: "synced",
   342→    objectType: "course",
   343→    objectId: "bulk",
   344→    reason: `Synced ${result.recordsRead} courses`,
   345→    metadata: {
   346→      created: result.recordsCreated,
   347→      updated: result.recordsUpdated,
   348→      skipped: result.recordsSkipped,
   349→      errors: result.errors.length,
   350→    },
   351→  });
   352→
   353→  return result;
   354→}

   497→export async function syncMemberships(params: SyncExecutionParams): Promise<SyncResult> {
   498→  const result: SyncResult = {
   499→    resource: "memberships",
   500→    recordsRead: 0,
   501→    recordsCreated: 0,
   502→    recordsUpdated: 0,
   503→    recordsSkipped: 0,
   504→    errors: [],
   505→    warnings: [],
   506→    cursor: null,
   507→  };
   508→
   509→  let cursor = params.cursor ?? null;
   510→  if (!cursor) {
   511→    const checkpoint = await getLatestCheckpoint(params.organizationId, "memberships");
   512→    if (checkpoint?.cursor) {
   513→      cursor = checkpoint.cursor;
   514→    }
   515→  }
   516→
   517→  let pagesProcessed = 0;
   518→
   519→  do {
   520→    const page = await params.providers.memberships.list({
   521→      companyId: params.companyId,
   522→      cursor,
   523→      pageSize: PAGE_SIZE,
   524→    });
   525→
   526→    result.recordsRead += page.items.length;
   527→    pagesProcessed++;
   528→
   529→    for (const membership of page.items) {
   530→      try {
   531→        // Normalize the external status — replaces `membership.status as any`
   532→        const { status: safeStatus, warning } = normalizeMembershipStatus(membership.status);
   533→        if (warning) {
   534→          result.warnings.push(warning);
   535→        }
   536→
   537→        // Find or create the student (upsert by composite key)
   538→        const student = await db.student.upsert({
   539→          where: {
   540→            organizationId_whopUserId: {
   541→              organizationId: params.organizationId,
   542→              whopUserId: membership.userId,
   543→            },
   544→          },
   545→          create: {
   546→            organizationId: params.organizationId,
   547→            whopUserId: membership.userId,
   548→          },
   549→          update: {},
   550→        });
   551→
   552→        // Find the product
   553→        const product = await db.product.findFirst({
   554→          where: {
   555→            organizationId: params.organizationId,
   556→            whopProductId: membership.productId,
   557→          },
   558→        });
   559→
   560→        if (!product) {
   561→          result.errors.push(`Membership ${membership.id}: product ${membership.productId} not found`);
   562→          result.recordsSkipped++;
   563→          continue;
   564→        }
   565→
   566→        // Check if membership already exists to determine create vs update
   567→        const existingMembership = await db.membership.findUnique({
   568→          where: { whopMembershipId: membership.id },
   569→          select: { id: true },
   570→        });
   571→
   572→        if (existingMembership) {
   573→          // UPDATE — increment recordsUpdated, NOT recordsCreated
   574→          await db.membership.update({
   575→            where: { id: existingMembership.id },
   576→            data: {
   577→              status: safeStatus,
   578→              renewalDate: membership.renewalDate ? new Date(membership.renewalDate) : null,
   579→              cancelledAt: membership.cancelledAt ? new Date(membership.cancelledAt) : null,
   580→              priceCents: membership.priceCents,
   581→              currency: membership.currency,
   582→              lastSyncedAt: new Date(),
   583→            },
   584→          });
   585→          result.recordsUpdated++;
   586→        } else {
   587→          // CREATE
   588→          await db.membership.create({
   589→            data: {
   590→              organizationId: params.organizationId,
   591→              studentId: student.id,
   592→              productId: product.id,
   593→              whopMembershipId: membership.id,
   594→              status: safeStatus,
   595→              joinedAt: new Date(membership.joinedAt),
   596→              renewalDate: membership.renewalDate ? new Date(membership.renewalDate) : null,
   597→              cancelledAt: membership.cancelledAt ? new Date(membership.cancelledAt) : null,
   598→              priceCents: membership.priceCents,
   599→              currency: membership.currency,
   600→              lastSyncedAt: new Date(),
   601→            },
   602→          });
   603→          result.recordsCreated++;
   604→        }
   605→      } catch (error) {
   606→        const failure = classifyCreateError(error, `membership-${membership.id}`);
   607→        if (failure.class === "duplicate_unique_constraint") {
   608→          // Idempotent — already ingested
   609→          result.recordsSkipped++;
   610→        } else {
   611→          result.errors.push(`Membership ${membership.id}: ${failure.message}`);
   612→          result.recordsSkipped++;
   613→        }
   614→      }
   615→    }
   616→
   617→    // Persist checkpoint
   618→    cursor = page.nextCursor;
   619→    if (params.syncExecutionId) {
   620→      await persistCheckpoint(
   621→        params.organizationId,
   622→        "memberships",
   623→        cursor,
   624→        null,
   625→        params.syncExecutionId,
   626→        pagesProcessed,
   627→      );
   628→    }
   629→  } while (cursor);
   630→
   631→  result.cursor = cursor;
   632→
   633→  await recordAuditEvent({
   634→    organizationId: params.organizationId,
   635→    action: "synced",
   636→    objectType: "membership",
   637→    objectId: "bulk",
   638→    reason: `Synced ${result.recordsRead} memberships`,
   639→    metadata: {
   640→      created: result.recordsCreated,
   641→      updated: result.recordsUpdated,
   642→      skipped: result.recordsSkipped,
   643→      warnings: result.warnings.length,
   644→    },
   645→  });
   646→
   647→  return result;
   648→}
   649→
   650→// ─── Course progress sync (Phase 10: idempotent + bounded) ──
   651→
   652→export async function syncCourseProgress(
   653→  params: SyncExecutionParams & { courseId: string },
   654→): Promise<SyncResult> {
   655→  const result: SyncResult = {
   656→    resource: "progress",
   657→    recordsRead: 0,
   658→    recordsCreated: 0,
   659→    recordsUpdated: 0,
   660→    recordsSkipped: 0,
   661→    errors: [],
   662→    warnings: [],
   663→    cursor: null,
   664→  };
   665→
   666→  // Find the internal course
   667→  const course = await db.course.findFirst({
   668→    where: {
   669→      organizationId: params.organizationId,
   670→      id: params.courseId,
   671→    },
   672→  });
   673→
   674→  if (!course) {
   675→    result.errors.push(`Course ${params.courseId} not found`);
   676→    return result;
   677→  }
   678→
   679→  let cursor = params.cursor ?? null;
   680→  if (!cursor) {
   681→    const checkpoint = await getLatestCheckpoint(params.organizationId, "progress");
   682→    if (checkpoint?.cursor) {
   683→      cursor = checkpoint.cursor;
   684→    }
   685→  }
   686→
   687→  let pagesProcessed = 0;
   688→  const now = new Date();
   689→
   690→  do {
   691→    const page = await params.providers.progress.listLessonInteractions({
   692→      companyId: params.companyId,
   693→      courseId: course.externalCourseId,
   694→      cursor,
   695→      pageSize: PAGE_SIZE,
   696→    });
   697→
   698→    result.recordsRead += page.items.length;
   699→    pagesProcessed++;
   700→
   701→    // Phase 10: Resolve all students for this page in one query
   702→    const userIds = [...new Set(page.items.map((i) => i.userId))];
   703→    const students = await db.student.findMany({
   704→      where: {
   705→        organizationId: params.organizationId,
   706→        whopUserId: { in: userIds },
   707→      },
   708→      select: { id: true, whopUserId: true },
   709→    });
   710→    const studentMap = new Map(students.map((s) => [s.whopUserId, s.id]));
   711→
   712→    // Phase 10: Check existing interaction IDs to split creates from skips
   713→    const interactionIds = page.items
   714→      .map((i) => i.id)
   715→      .filter(Boolean);
   716→    const existingEvents = interactionIds.length > 0
   717→      ? await db.progressEvent.findMany({
   718→          where: {
   719→            organizationId: params.organizationId,
   720→            externalInteractionId: { in: interactionIds },
   721→          },
   722→          select: { externalInteractionId: true },
   723→        })
   724→      : [];
   725→    const existingInteractionSet = new Set(existingEvents.map((e) => e.externalInteractionId));
   726→
   727→    // Phase 10: Also check existing payload hashes for dedup
   728→    // and fetch sourceTimestamp so we can handle out-of-order / older-after-newer
   729→    const payloadHashes = page.items.map((interaction) =>
   730→      computePayloadHash({
   731→        userId: interaction.userId,
   732→        courseId: interaction.courseId,
   733→        lessonId: interaction.lessonId,
   734→        action: interaction.completed ? "completed" : "started",
   735→        sourceTimestamp: interaction.sourceTimestamp,
   736→      }),
   737→    );
   738→    const existingByHash = payloadHashes.length > 0
   739→      ? await db.progressEvent.findMany({
   740→          where: {
   741→            organizationId: params.organizationId,
   742→            payloadHash: { in: payloadHashes },
   743→          },
   744→          select: { payloadHash: true, sourceTimestamp: true, id: true },
   745→        })
   746→      : [];
   747→    const existingHashSet = new Set(existingByHash.map((e) => e.payloadHash));
   748→    // Map: payloadHash → existing event for out-of-order update detection
   749→    const existingByHashMap = new Map(existingByHash.map((e) => [e.payloadHash, e]));
   750→
   751→    // Build create data, filtering out duplicates and missing students
   752→    const toCreate: Array<{
   753→      organizationId: string;
   754→      studentId: string;
   755→      courseId: string;
   756→      externalInteractionId: string;
   757→      lessonId: string;
   758→      lessonIndex: number;
   759→      lessonTitle: string | null;
   760→      action: string;
   761→      payloadHash: string;
   762→      sourceTimestamp: Date;
   763→      receivedAt: Date;
   764→      sourceVersion: string | null;
   765→      occurredAt: Date;
   766→    }> = [];
   767→
   768→    for (const interaction of page.items) {
   769→      // Skip if student not found — students with activity but no qualifying membership
   770→      const studentId = studentMap.get(interaction.userId);
   771→      if (!studentId) {
   772→        result.warnings.push(
   773→          `Interaction ${interaction.id}: student ${interaction.userId} not found — skipped (membership may arrive later)`,
   774→        );
   775→        result.recordsSkipped++;
   776→        continue;
   777→      }
   778→
   779→      // Phase 10: Skip duplicates by external interaction ID
   780→      if (existingInteractionSet.has(interaction.id)) {
   781→        result.recordsSkipped++;
   782→        continue;
   783→      }
   784→
   785→      // Phase 10: Compute payload hash
   786→      const hash = computePayloadHash({
   787→        userId: interaction.userId,
   788→        courseId: interaction.courseId,
   789→        lessonId: interaction.lessonId,
   790→        action: interaction.completed ? "completed" : "started",
   791→        sourceTimestamp: interaction.sourceTimestamp,
   792→      });
   793→
   794→      // Phase 10: If a different payloadHash exists for the same interaction ID,
   795→      // an older event is being replaced by a newer one — update the source timestamp
   796→      const existingByHash = existingByHashMap.get(hash);
   797→      if (existingHashSet.has(hash)) {
   798→        // Check if the incoming event has a newer source timestamp than stored
   799→        const incomingSource = new Date(interaction.sourceTimestamp);
   800→        if (
   801→          existingByHash?.sourceTimestamp &&
   802→          incomingSource > existingByHash.sourceTimestamp
   803→        ) {
   804→          // Out-of-order: newer event for same dedup key — update timestamp
   805→          try {
   806→            await db.progressEvent.update({
   807→              where: { id: existingByHash.id },
   808→              data: {
   809→                sourceTimestamp: incomingSource,
   810→                occurredAt: new Date(interaction.createdAt),
   811→                receivedAt: now,
   812→              },
   813→            });
   814→            result.recordsUpdated++;
   815→          } catch {
   816→            // Race condition — skip gracefully
   817→            result.recordsSkipped++;
   818→          }
   819→        } else {
   820→          // Exact duplicate or older event — skip
   821→          result.recordsSkipped++;
   822→        }
   823→        continue;
   824→      }
   825→
   826→      toCreate.push({
   827→        organizationId: params.organizationId,
   828→        studentId,
   829→        courseId: course.id,
   830→        externalInteractionId: interaction.id,
   831→        lessonId: interaction.lessonId,
   832→        lessonIndex: 0, // Lesson index not available from provider; use 0 as placeholder
   833→        lessonTitle: interaction.lessonTitle,
   834→        action: interaction.completed ? "completed" : "started",
   835→        payloadHash: hash,
   836→        sourceTimestamp: new Date(interaction.sourceTimestamp),
   837→        receivedAt: now,
   838→        sourceVersion: null, // No version from current provider contract
   839→        occurredAt: new Date(interaction.createdAt),
   840→      });
   841→    }
   842→
   843→    // Phase 10: Batch create with skipDuplicates for true idempotency
   844→    if (toCreate.length > 0) {
   845→      // Process in sub-batches to avoid oversized queries
   846→      for (let i = 0; i < toCreate.length; i += PROGRESS_BATCH_SIZE) {
   847→        const batch = toCreate.slice(i, i + PROGRESS_BATCH_SIZE);
   848→        try {
   849→          const createResult = await db.progressEvent.createMany({
   850→            data: batch,
   851→            skipDuplicates: true,
   852→          });
   853→          result.recordsCreated += createResult.count;
   854→          // Any items not created were duplicates (skipDuplicates silently skips)
   855→          result.recordsSkipped += batch.length - createResult.count;
   856→        } catch (error) {
   857→          const failure = classifyCreateError(error, "progress-batch");
   858→          if (
   859→            failure.class === "duplicate_unique_constraint" ||
   860→            failure.class === "duplicate_payload_hash"
   861→          ) {
   862→            // These are expected under race conditions — all treated as skipped
   863→            result.recordsSkipped += batch.length;
   864→          } else {
   865→            result.errors.push(`Progress batch create: ${failure.message}`);
   866→            result.recordsSkipped += batch.length;
   867→          }
   868→        }
   869→      }
   870→    }
   871→
   872→    // Persist checkpoint
   873→    cursor = page.nextCursor;
   874→    if (params.syncExecutionId) {
   875→      await persistCheckpoint(
   876→        params.organizationId,
   877→        "progress",
   878→        cursor,
   879→        null,
   880→        params.syncExecutionId,
   881→        pagesProcessed,
   882→      );
   883→    }
   884→  } while (cursor);
   885→
   886→  // Phase 10: Recalculate course states using bounded aggregation
   887→  await recalculateCourseStatesBounded(params.organizationId, course.id);
   888→
   889→  result.cursor = cursor;
   890→
   891→  await recordAuditEvent({
   892→    organizationId: params.organizationId,
   893→    action: "synced",
   894→    objectType: "progress",
   895→    objectId: course.id,
   896→    reason: `Synced ${result.recordsRead} progress events`,
   897→    metadata: {
   898→      created: result.recordsCreated,
   899→      skipped: result.recordsSkipped,
   900→      duplicatesHandled: result.recordsSkipped,
   901→    },
   902→  });
   903→
   904→  return result;
   905→}
   906→
   907→// ─── Reconciliation (Phase 11: set-based queries) ───────────
   908→
   909→export interface ReconciliationResult {
   910→  matched: number;
   911→  membershipWithoutCourseActivity: number;
   912→  courseActivityWithoutMembership: number;
   913→  unmappedProduct: number;
   914→  missingSourceFields: number;
   915→  staleSourceRecord: number;
   916→  totalEvaluated: number;
   917→  pagesProcessed: number;
   918→}
   919→
   920→export async function reconcile(
   921→  organizationId: string,
   922→  courseId: string,
   923→  syncExecutionId?: string,
   924→  pageLimit?: number, // Phase 11: pagination — max pages to process
   925→): Promise<ReconciliationResult> {
   926→  const result: ReconciliationResult = {
   927→    matched: 0,
   928→    membershipWithoutCourseActivity: 0,
   929→    courseActivityWithoutMembership: 0,
   930→    unmappedProduct: 0,
   931→    missingSourceFields: 0,
   932→    staleSourceRecord: 0,
   933→    totalEvaluated: 0,
   934→    pagesProcessed: 0,
   935→  };
   936→
   937→  // Create a reconciliation run record
   938→  const run = await createReconciliationRun(organizationId, courseId, syncExecutionId);
   939→
   940→  // Phase 11: Set-based queries instead of per-student N+1
   941→
   942→  // 1. Get all confirmed mappings for this course (bounded — typically ≤10 products per course)
   943→  const mappings = await db.productCourseMapping.findMany({
   944→    where: { organizationId, courseId, isConfirmed: true },
   945→    select: { id: true, productId: true, courseId: true },
   946→  });
   947→
   948→  const mappedProductIds = new Set(mappings.map((m) => m.productId));
   949→
   950→  // 2. Get all student IDs with qualifying memberships for mapped products (ONE query)
   951→  const studentsWithMembership = await db.membership.findMany({
   952→    where: {
   953→      organizationId,
   954→      productId: { in: Array.from(mappedProductIds) },
   955→    },
   956→    select: { studentId: true, id: true, productId: true, status: true },
   957→  });
   958→
   959→  // Build map: studentId → membership info for classification
   960→  const membershipByStudent = new Map<string, { membershipId: string; productId: string; status: string }[]>();
   961→  for (const m of studentsWithMembership) {
   962→    const existing = membershipByStudent.get(m.studentId) ?? [];
   963→    existing.push({ membershipId: m.id, productId: m.productId, status: m.status });
   964→    membershipByStudent.set(m.studentId, existing);
   965→  }
   966→
   967→  const studentIdsWithMembership = new Set(membershipByStudent.keys());
   968→
   969→  // 3. Get all student IDs with course activity (ONE query)
   970→  const courseStates = await db.studentCourseState.findMany({
   971→    where: { organizationId, courseId },
   972→    select: { studentId: true, id: true, lessonsCompleted: true, totalLessons: true, lastSyncedAt: true },
   973→  });
   974→
   975→  const studentIdsWithActivity = new Set(courseStates.map((s) => s.studentId));
   976→
   977→  // Phase 11: Detect students with memberships for unmapped products
   978→  const allMembershipsForOrg = await db.membership.findMany({
   979→    where: {
   980→      organizationId,
   981→      status: { in: ["active", "trialing"] },
   982→    },
   983→    select: { studentId: true, id: true, productId: true },
   984→  });
   985→  const studentsWithUnmappedProduct = allMembershipsForOrg.filter(
   986→    (m) => !mappedProductIds.has(m.productId),
   987→  );
   988→  const unmappedStudentIds = new Set(studentsWithUnmappedProduct.map((m) => m.studentId));
   989→
   990→  // Phase 11: Detect stale source records (lastSyncedAt > 24h ago)
   991→  const STALE_THRESHOLD_MS = 24 * 60 * 60 * 1000;
   992→  const now = new Date();
   993→  const staleStudentIds = new Set(
   994→    courseStates
   995→      .filter((s) => s.lastSyncedAt && (now.getTime() - s.lastSyncedAt.getTime()) > STALE_THRESHOLD_MS)
   996→      .map((s) => s.studentId),
   997→  );
   998→
   999→  // 4. Set-based classification — no per-student queries
  1000→  // Matched: students in both sets
  1001→  const matchedStudentIds = new Set(
  1002→    [...studentIdsWithMembership].filter((id) => studentIdsWithActivity.has(id)),
  1003→  );
  1004→
  1005→  // Membership without course activity
  1006→  const membershipOnlyStudentIds = new Set(
  1007→    [...studentIdsWithMembership].filter((id) => !studentIdsWithActivity.has(id)),
  1008→  );
  1009→
  1010→  // Course activity without membership (retroactive link opportunity)
  1011→  const activityOnlyStudentIds = new Set(
  1012→    [...studentIdsWithActivity].filter((id) => !studentIdsWithMembership.has(id)),
  1013→  );
  1014→
  1015→  result.matched = matchedStudentIds.size;
  1016→  result.membershipWithoutCourseActivity = membershipOnlyStudentIds.size;
  1017→  result.courseActivityWithoutMembership = activityOnlyStudentIds.size;
  1018→  result.unmappedProduct = [...unmappedStudentIds].filter(
  1019→    (id) => !studentIdsWithMembership.has(id),
  1020→  ).length;
  1021→  result.staleSourceRecord = staleStudentIds.size;
  1022→  result.totalEvaluated = studentIdsWithMembership.size + activityOnlyStudentIds.size;
  1023→
  1024→  // 5. Persist reconciliation outcomes in paginated batches
  1025→  const outcomeBatches: Array<{
  1026→    reconciliationRunId: string;
  1027→    organizationId: string;
  1028→    studentId: string;
  1029→    membershipId: string | null;
  1030→    courseId: string;
  1031→    mappingId: string | null;
  1032→    classification: string;
  1033→    evidenceJson: unknown;
  1034→  }> = [];
  1035→
  1036→  // Matched outcomes
  1037→  for (const studentId of matchedStudentIds) {
  1038→    const memberships = membershipByStudent.get(studentId) ?? [];
  1039→    const mappingId = mappings.find((m) =>
  1040→      memberships.some((mem) => mem.productId === m.productId),
  1041→    )?.id ?? null;
  1042→    outcomeBatches.push({
  1043→      reconciliationRunId: run.id,
  1044→      organizationId,
  1045→      studentId,
  1046→      membershipId: memberships[0]?.membershipId ?? null,
  1047→      courseId,
  1048→      mappingId,
  1049→      classification: "matched",
  1050→      evidenceJson: { hasMembership: true, hasCourseActivity: true, isStale: staleStudentIds.has(studentId) },
  1051→    });
  1052→  }
  1053→
  1054→  // Membership without course activity
  1055→  for (const studentId of membershipOnlyStudentIds) {
  1056→    const memberships = membershipByStudent.get(studentId) ?? [];
  1057→    const mappingId = mappings.find((m) =>
  1058→      memberships.some((mem) => mem.productId === m.productId),
  1059→    )?.id ?? null;
  1060→    outcomeBatches.push({
  1061→      reconciliationRunId: run.id,
  1062→      organizationId,
  1063→      studentId,
  1064→      membershipId: memberships[0]?.membershipId ?? null,
  1065→      courseId,
  1066→      mappingId,
  1067→      classification: "membership_without_course_activity",
  1068→      evidenceJson: { hasMembership: true, hasCourseActivity: false, isStale: staleStudentIds.has(studentId) },
  1069→    });
  1070→  }
  1071→
  1072→  // Course activity without membership
  1073→  for (const studentId of activityOnlyStudentIds) {
  1074→    outcomeBatches.push({
  1075→      reconciliationRunId: run.id,
  1076→      organizationId,
  1077→      studentId,
  1078→      membershipId: null,
  1079→      courseId,
  1080→      mappingId: null,
  1081→      classification: "course_activity_without_membership",
  1082→      evidenceJson: { hasMembership: false, hasCourseActivity: true, isStale: staleStudentIds.has(studentId) },
  1083→    });
  1084→  }
  1085→
  1086→  // Unmapped product outcomes
  1087→  for (const studentId of unmappedStudentIds) {
  1088→    if (studentIdsWithMembership.has(studentId)) continue; // Already classified above
  1089→    const membership = allMembershipsForOrg.find((m) => m.studentId === studentId);
  1090→    outcomeBatches.push({
  1091→      reconciliationRunId: run.id,
  1092→      organizationId,
  1093→      studentId,
  1094→      membershipId: membership?.id ?? null,
  1095→      courseId,
  1096→      mappingId: null,
  1097→      classification: "unmapped_product",
  1098→      evidenceJson: { hasMembership: true, hasCourseActivity: studentIdsWithActivity.has(studentId), productId: membership?.productId },
  1099→    });
  1100→  }
  1101→
  1102→  // Stale source record outcomes
  1103→  for (const studentId of staleStudentIds) {
  1104→    // Stale records may overlap with other classifications — add separate stale outcome
  1105→    const state = courseStates.find((s) => s.studentId === studentId);
  1106→    if (state) {
  1107→      outcomeBatches.push({
  1108→        reconciliationRunId: run.id,
  1109→        organizationId,
  1110→        studentId,
  1111→        membershipId: null,
  1112→        courseId,
  1113→        mappingId: null,
  1114→        classification: "stale_source_record",
  1115→        evidenceJson: { lastSyncedAt: state.lastSyncedAt?.toISOString() },
  1116→      });
  1117→    }
  1118→  }
  1119→
  1120→  // Batch persist outcomes with pagination
  1121→  const maxPages = pageLimit ?? Infinity;
  1122→  for (let i = 0; i < outcomeBatches.length; i += RECONCILIATION_PAGE_SIZE) {
  1123→    if (result.pagesProcessed >= maxPages) break;
  1124→    const batch = outcomeBatches.slice(i, i + RECONCILIATION_PAGE_SIZE);
  1125→    await db.reconciliationOutcome.createMany({
  1126→      data: batch.map((o) => ({
  1127→        reconciliationRunId: o.reconciliationRunId,
  1128→        organizationId: o.organizationId,
  1129→        studentId: o.studentId,
  1130→        membershipId: o.membershipId,
  1131→        courseId: o.courseId,
  1132→        mappingId: o.mappingId,
  1133→        classification: o.classification as Prisma.InputJsonValue,
  1134→        evidenceJson: o.evidenceJson as Prisma.InputJsonValue,
  1135→      })),
  1136→    });
  1137→    result.pagesProcessed++;
  1138→  }
  1139→
  1140→  // Complete the reconciliation run record
  1141→  await completeReconciliationRun(run.id, {
  1142→    matched: result.matched,
  1143→    membershipWithoutCourseActivity: result.membershipWithoutCourseActivity,
  1144→    courseActivityWithoutMembership: result.courseActivityWithoutMembership,
  1145→    unmappedProduct: result.unmappedProduct,
  1146→    missingSourceFields: result.missingSourceFields,
  1147→    staleSourceRecord: result.staleSourceRecord,
  1148→  });
  1149→
  1150→  return result;
  1151→}

  1153→// ─── Candidate detection (Phase 12: full eligibility checks) ─
  1154→
  1155→export interface CandidateDetectionResult {
  1156→  candidatesFound: number;
  1157→  snapshotsCreated: number;
  1158→  snapshotsSkipped: number;
  1159→  errors: string[];
  1160→  warnings: string[];
  1161→  checksPerformed: number;
  1162→}
  1163→
  1164→/**
  1165→ * Detect candidates for an Activation Rescue campaign.
  1166→ *
  1167→ * Phase 12: Operates ONLY on the campaign's confirmed mapping.
  1168→ * All 17 eligibility checks are verified before creating a snapshot.
  1169→ * Uses batch queries — no N+1 patterns.
  1170→ */
  1171→export async function detectCandidates(
  1172→  organizationId: string,
  1173→  campaignId: string,
  1174→): Promise<CandidateDetectionResult> {
  1175→  const result: CandidateDetectionResult = {
  1176→    candidatesFound: 0,
  1177→    snapshotsCreated: 0,
  1178→    snapshotsSkipped: 0,
  1179→    errors: [],
  1180→    warnings: [],
  1181→    checksPerformed: 0,
  1182→  };
  1183→
  1184→  const now = new Date();
  1185→
  1186→  // ─── Pre-flight checks (fail fast) ────────────────────────
  1187→
  1188→  // Fetch campaign with organization and latest version
  1189→  const campaign = await db.campaign.findFirst({
  1190→    where: { id: campaignId, organizationId },
  1191→    include: {
  1192→      organization: {
  1193→        include: { installations: true },
  1194→      },
  1195→      versions: { orderBy: { versionNumber: "desc" }, take: 1 },
  1196→      confirmedMapping: {
  1197→        include: {
  1198→          product: true,
  1199→          course: true,
  1200→        },
  1201→      },
  1202→    },
  1203→  });
  1204→
  1205→  // Check: Campaign exists and is Activation Rescue
  1206→  if (!campaign || campaign.type !== "activation_rescue") {
  1207→    result.errors.push("Campaign not found or not an Activation Rescue campaign");
  1208→    return result;
  1209→  }
  1210→
  1211→  const latestVersion = campaign.versions[0];
  1212→  const org = campaign.organization;
  1213→
  1214→  // Check 1: Organisation active
  1215→  if (org.status !== "active") {
  1216→    result.errors.push(`Organisation status is "${org.status}", not "active"`);
  1217→    return result;
  1218→  }
  1219→
  1220→  // Check 2: Organisation not paused
  1221→  if (org.isPaused) {
  1222→    result.errors.push("Organisation is paused");
  1223→    return result;
  1224→  }
  1225→
  1226→  // Check 3: Installation active
  1227→  const activeInstallation = org.installations.find((i) => i.status === "active");
  1228→  if (!activeInstallation) {
  1229→    result.errors.push("No active installation");
  1230→    return result;
  1231→  }
  1232→
  1233→  // Check 4: Campaign active
  1234→  if (campaign.status !== "active") {
  1235→    result.errors.push(`Campaign status is "${campaign.status}", not "active"`);
  1236→    return result;
  1237→  }
  1238→
  1239→  // Check 5: Campaign is Activation Rescue (already verified above)
  1240→  // Check 6: Manual approval enabled
  1241→  if (campaign.approvalMode !== "manual") {
  1242→    result.errors.push("Campaign does not have manual approval enabled");
  1243→    return result;
  1244→  }
  1245→
  1246→  // Check 7: Campaign version exists
  1247→  if (!latestVersion) {
  1248→    result.errors.push("No campaign version exists");
  1249→    return result;
  1250→  }
  1251→
  1252→  // Check 8: Confirmed mapping belongs to campaign
  1253→  const mapping = campaign.confirmedMapping;
  1254→  if (!mapping) {
  1255→    result.errors.push("Campaign has no confirmed mapping");
  1256→    return result;
  1257→  }
  1258→
  1259→  if (!mapping.isConfirmed) {
  1260→    result.errors.push("Campaign's mapping is not confirmed");
  1261→    return result;
  1262→  }
  1263→
  1264→  const courseId = mapping.courseId;
  1265→  const productId = mapping.productId;
  1266→
  1267→  // ─── Batch queries (no N+1) ───────────────────────────────
  1268→
  1269→  // All memberships for the mapped product that are active or trialing
  1270→  const qualifyingMemberships = await db.membership.findMany({
  1271→    where: {
  1272→      organizationId,
  1273→      productId,
  1274→      status: { in: ["active", "trialing"] },
  1275→    },
  1276→    include: { student: true },
  1277→  });
  1278→
  1279→  if (qualifyingMemberships.length === 0) {
  1280→    return result; // No candidates possible
  1281→  }
  1282→
  1283→  const candidateStudentIds = new Set(qualifyingMemberships.map((m) => m.studentId));
  1284→
  1285→  // Batch: course activity states for all candidate students
  1286→  const courseStates = await db.studentCourseState.findMany({
  1287→    where: {
  1288→      organizationId,
  1289→      courseId,
  1290→      studentId: { in: Array.from(candidateStudentIds) },
  1291→    },
  1292→  });
  1293→  const activityByStudent = new Map(courseStates.map((s) => [s.studentId, s]));
  1294→
  1295→  // Batch: suppressions for all candidate students (course, campaign, and org level)
  1296→  const suppressions = await db.suppression.findMany({
  1297→    where: {
  1298→      organizationId,
  1299→      studentId: { in: Array.from(candidateStudentIds) },
  1300→    },
  1301→  });
  1302→  const suppressedStudentIds = new Set(suppressions.map((s) => s.studentId));
  1303→
  1304→  // Batch: existing active interventions for these students on this campaign
  1305→  const activeInterventions = await db.intervention.findMany({
  1306→    where: {
  1307→      organizationId,
  1308→      studentId: { in: Array.from(candidateStudentIds) },
  1309→      campaignId,
  1310→      state: { notIn: ["dismissed", "stopped", "failed"] },
  1311→    },
  1312→  });
  1313→  const studentsWithActiveIntervention = new Set(activeInterventions.map((i) => i.studentId));
  1314→
  1315→  // Batch: last intervention per student for cooldown check
  1316→  const lastInterventions = await db.intervention.findMany({
  1317→    where: {
  1318→      organizationId,
  1319→      studentId: { in: Array.from(candidateStudentIds) },
  1320→      campaignId,
  1321→    },
  1322→    orderBy: { createdAt: "desc" },
  1323→    // We'll process these to get the latest per student
  1324→  });
  1325→  const lastInterventionByStudent = new Map<string, Date>();
  1326→  for (const iv of lastInterventions) {
  1327→    if (!lastInterventionByStudent.has(iv.studentId)) {
  1328→      lastInterventionByStudent.set(iv.studentId, iv.createdAt);
  1329→    }
  1330→  }
  1331→
  1332→  // Batch: recent intervention counts per student (for per-student message limit)
  1333→  const thirtyDaysAgo = new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000);
  1334→  const recentInterventions = await db.intervention.findMany({
  1335→    where: {
  1336→      organizationId,
  1337→      studentId: { in: Array.from(candidateStudentIds) },
  1338→      campaignId,
  1339→      createdAt: { gte: thirtyDaysAgo },
  1340→      state: { in: ["notification_accepted", "delivered"] },
  1341→    },
  1342→    select: { studentId: true },
  1343→  });
  1344→  const recentInterventionCounts = new Map<string, number>();
  1345→  for (const iv of recentInterventions) {
  1346→    recentInterventionCounts.set(iv.studentId, (recentInterventionCounts.get(iv.studentId) ?? 0) + 1);
  1347→  }
  1348→
  1349→  // Batch: org-wide message count in last 30 days
  1350→  const orgWideMessageCount = await db.intervention.count({
  1351→    where: {
  1352→      organizationId,
  1353→      createdAt: { gte: thirtyDaysAgo },
  1354→      state: { in: ["notification_accepted", "delivered"] },
  1355→    },
  1356→  });
  1357→
  1358→  // Batch: campaign message count in last 30 days
  1359→  const campaignMessageCount = await db.intervention.count({
  1360→    where: {
  1361→      organizationId,
  1362→      campaignId,
  1363→      createdAt: { gte: thirtyDaysAgo },
  1364→      state: { in: ["notification_accepted", "delivered"] },
  1365→    },
  1366→  });
  1367→
  1368→  // Batch: existing eligibility snapshots for this campaign version + window
  1369→  // (to prevent duplicate snapshots on re-run)
  1370→  const eligibilityWindowStart = new Date(
  1371→    now.getFullYear(),
  1372→    now.getMonth(),
  1373→    now.getDate(),
  1374→  ); // Start of today as the eligibility window
  1375→
  1376→  const existingSnapshots = await db.eligibilitySnapshot.findMany({
  1377→    where: {
  1378→      organizationId,
  1379→      studentId: { in: Array.from(candidateStudentIds) },
  1380→      campaignVersionId: latestVersion.id,
  1381→      eligibilityWindowStart,
  1382→    },
  1383→    select: { studentId: true },
  1384→  });
  1385→  const studentsWithExistingSnapshot = new Set(existingSnapshots.map((s) => s.studentId));
  1386→
  1387→  // Batch: check plan allows monitored members
  1388→  const entitlement = await db.subscriptionEntitlement.findFirst({
  1389→    where: { organizationId },
  1390→  });
  1391→  const plan = entitlement
  1392→    ? await db.plan.findUnique({ where: { tier: entitlement.planTier } })
  1393→    : null;
  1394→  const maxMonitoredMembers = plan?.maxMonitoredMembers ?? Infinity;
  1395→
  1396→  // Count current monitored members
  1397→  const currentMonitoredMembers = await db.membership.count({
  1398→    where: {
  1399→      organizationId,
  1400→      status: { in: ["active", "trialing"] },
  1401→    },
  1402→  });
  1403→
  1404→  // Batch: source data freshness check
  1405→  const latestCheckpoint = await db.syncCheckpoint.findFirst({
  1406→    where: { organizationId, resource: "memberships" },
  1407→    orderBy: { updatedAt: "desc" },
  1408→  });
  1409→  const SOURCE_FRESHNESS_MAX_AGE_MS = 24 * 60 * 60 * 1000; // 24 hours
  1410→  const sourceDataFresh = latestCheckpoint
  1411→    ? (now.getTime() - latestCheckpoint.updatedAt.getTime()) < SOURCE_FRESHNESS_MAX_AGE_MS
  1412→    : false;
  1413→
  1414→  // ─── Evaluate each candidate ──────────────────────────────
  1415→
  1416→  // Process in batches to avoid memory pressure
  1417→  const candidateArray = Array.from(qualifyingMemberships);
  1418→
  1419→  for (let batchStart = 0; batchStart < candidateArray.length; batchStart += CANDIDATE_BATCH_SIZE) {
  1420→    const batch = candidateArray.slice(batchStart, batchStart + CANDIDATE_BATCH_SIZE);
  1421→
  1422→    for (const membership of batch) {
  1423→      result.checksPerformed++;
  1424→      const studentId = membership.studentId;
  1425→
  1426→      // Check 9: Membership belongs to mapped product (guaranteed by our query)
  1427→
  1428→      // Check 10: Membership active or trialing (guaranteed by our query)
  1429→
  1430→      // Check 11: Membership not ending (no renewalDate or renewalDate in the future)
  1431→      if (membership.renewalDate && membership.renewalDate <= now) {
  1432→        continue;
  1433→      }
  1434→
  1435→      // Check 12: Activation delay elapsed
  1436→      const daysSinceJoin = (now.getTime() - membership.joinedAt.getTime()) / (1000 * 60 * 60 * 24);
  1437→      if (daysSinceJoin < mapping.activationDelayDays) {
  1438→        continue;
  1439→      }
  1440→
  1441→      // Check 13: Course activity absent
  1442→      const courseState = activityByStudent.get(studentId);
  1443→      if (courseState && courseState.lessonsCompleted > 0) {
  1444→        continue;
  1445→      }
  1446→
  1447→      // Check 14: No course-, campaign- or organisation-level suppression
  1448→      if (suppressedStudentIds.has(studentId)) {
  1449→        continue;
  1450→      }
  1451→
  1452→      // Check 15: No existing equivalent active intervention
  1453→      if (studentsWithActiveIntervention.has(studentId)) {
  1454→        continue;
  1455→      }
  1456→
  1457→      // Campaign cooldown check
  1458→      const lastIvDate = lastInterventionByStudent.get(studentId);
  1459→      if (lastIvDate) {
  1460→        const daysSinceLast = (now.getTime() - lastIvDate.getTime()) / (1000 * 60 * 60 * 24);
  1461→        if (daysSinceLast < campaign.cooldownDays) {
  1462→          continue;
  1463→        }
  1464→      }
  1465→
  1466→      // Check: Per-student message limit
  1467→      const studentRecentCount = recentInterventionCounts.get(studentId) ?? 0;
  1468→      if (studentRecentCount >= campaign.maxMessagesPerStudent) {
  1469→        continue;
  1470→      }
  1471→
  1472→      // Check: Org-wide message limit
  1473→      if (orgWideMessageCount >= campaign.maxMessagesPerOrg) {
  1474→        continue;
  1475→      }
  1476→
  1477→      // Check 18: Campaign message limit clear
  1478→      if (campaignMessageCount >= campaign.maxMessagesPerStudent) {
  1479→        continue;
  1480→      }
  1481→
  1482→      // Check 16: Plan allows monitored member
  1483→      if (currentMonitoredMembers >= maxMonitoredMembers) {
  1484→        continue;
  1485→      }
  1486→
  1487→      // Check 20: Source data sufficiently fresh
  1488→      if (!sourceDataFresh) {
  1489→        result.warnings.push("Source data is stale — skipping candidate detection");
  1490→        continue;
  1491→      }
  1492→
  1493→      // Check: Existing snapshot for this window (idempotency)
  1494→      if (studentsWithExistingSnapshot.has(studentId)) {
  1495→        result.snapshotsSkipped++;
  1496→        continue;
  1497→      }
  1498→
  1499→      // ─── Candidate found — create immutable eligibility snapshot ───
  1500→      result.candidatesFound++;
  1501→
  1502→      const evidence = {
  1503→        detectedAt: now.toISOString(),
  1504→        membershipId: membership.id,
  1505→        membershipStatus: membership.status,
  1506→        joinedAt: membership.joinedAt.toISOString(),
  1507→        daysSinceJoin: Math.floor(daysSinceJoin),
  1508→        activationDelayDays: mapping.activationDelayDays,
  1509→        courseId: mapping.courseId,
  1510→        productId: mapping.productId,
  1511→        campaignId,
  1512→        campaignVersionId: latestVersion.id,
  1513→        checks: {
  1514→          orgActive: true,
  1515→          orgNotPaused: true,
  1516→          installationActive: true,
  1517→          campaignActive: true,
  1518→          campaignIsActivationRescue: true,
  1519→          manualApprovalEnabled: true,
  1520→          campaignVersionExists: true,
  1521→          confirmedMappingBelongsToCampaign: true,
  1522→          membershipBelongsToMappedProduct: true,
  1523→          membershipActiveOrTrialing: true,
  1524→          membershipNotEnding: !membership.renewalDate || membership.renewalDate > now,
  1525→          activationDelayElapsed: true,
  1526→          courseActivityAbsent: true,
  1527→          notSuppressed: true,
  1528→          noActiveIntervention: true,
  1529→          campaignCooldownClear: !lastIvDate || (now.getTime() - lastIvDate.getTime()) / (1000 * 60 * 60 * 24) >= campaign.cooldownDays,
  1530→          orgMessageLimitClear: orgWideMessageCount < campaign.maxMessagesPerOrg,
  1531→          campaignMessageLimitClear: campaignMessageCount < campaign.maxMessagesPerStudent,
  1532→          planAllowsMonitoredMember: currentMonitoredMembers < maxMonitoredMembers,
  1533→          sourceDataFresh: true,
  1534→        },
  1535→      };
  1536→
  1537→      // Build idempotency key: student + campaign version + eligibility window
  1538→      const idempotencyKey = `${studentId}:${latestVersion.id}:${eligibilityWindowStart.toISOString()}`;
  1539→
  1540→      try {
  1541→        await db.eligibilitySnapshot.create({
  1542→          data: {
  1543→            organizationId,
  1544→            studentId,
  1545→            campaignId,
  1546→            campaignVersionId: latestVersion.id,
  1547→            idempotencyKey,
  1548→            state: "eligible",
  1549→            evidenceJson: evidence as Prisma.InputJsonValue,
  1550→            detectedAt: now,
  1551→            eligibilityWindowStart,
  1552→            expiresAt: new Date(now.getTime() + 7 * 24 * 60 * 60 * 1000),
  1553→          },
  1554→        });
  1555→        result.snapshotsCreated++;
  1556→      } catch (error) {
  1557→        const failure = classifyCreateError(error, idempotencyKey);
  1558→        if (
  1559→          failure.class === "duplicate_unique_constraint" ||
  1560→          failure.class === "duplicate_payload_hash"
  1561→        ) {
  1562→          // Idempotent — snapshot already exists for this window
  1563→          result.snapshotsSkipped++;
  1564→        } else {
  1565→          result.errors.push(
  1566→            `Student ${studentId}: ${failure.message}`,
  1567→          );
  1568→        }
  1569→      }
  1570→    }
  1571→  }
  1572→
  1573→  return result;
  1574→}

  1578→/**
  1579→ * Run a full sync with durable execution records and stage tracking.
  1580→ * Returns the sync execution ID for observability.
  1581→ */
  1582→export async function runFullSync(
  1583→  params: SyncExecutionParams & { resources?: string[] },
  1584→): Promise<{ executionId: string; results: SyncResult[] }> {
  1585→  const resources = params.resources ?? ["courses", "products", "memberships"];
  1586→  const results: SyncResult[] = [];
  1587→
  1588→  // Create the execution record
  1589→  const execution = await createSyncExecution({
  1590→    orgId: params.organizationId,
  1591→    trigger: params.requestedBy ? "manual" : "scheduled",
  1592→    requestedBy: params.requestedBy,
  1593→    jobId: params.syncExecutionId,
  1594→  });
  1595→
  1596→  try {
  1597→    for (const resource of resources) {
  1598→      const stage = await createSyncStage(execution.id, resource);
  1599→
  1600→      let syncResult: SyncResult;
  1601→      switch (resource) {
  1602→        case "courses":
  1603→          syncResult = await syncCourses({ ...params, syncExecutionId: execution.id });
  1604→          break;
  1605→        case "products":
  1606→          syncResult = await syncProducts({ ...params, syncExecutionId: execution.id });
  1607→          break;
  1608→        case "memberships":
  1609→          syncResult = await syncMemberships({ ...params, syncExecutionId: execution.id });
  1610→          break;
  1611→        default:
  1612→          await completeSyncStage(stage.id, {
  1613→            pagesProcessed: 0, recordsRead: 0, recordsCreated: 0,
  1614→            recordsUpdated: 0, recordsSkipped: 0,
  1615→          }, "skipped");
  1616→          continue;
  1617→      }
  1618→
  1619→      results.push(syncResult);
  1620→
  1621→      // Complete the stage with counts
  1622→      await completeSyncStage(stage.id, {
  1623→        pagesProcessed: 0, // Pages are tracked inside the sync functions
  1624→        recordsRead: syncResult.recordsRead,
  1625→        recordsCreated: syncResult.recordsCreated,
  1626→        recordsUpdated: syncResult.recordsUpdated,
  1627→        recordsSkipped: syncResult.recordsSkipped,
  1628→      });
  1629→    }
  1630→
  1631→    await completeSyncExecution(execution.id, "completed");
  1632→  } catch (error) {
  1633→    await completeSyncExecution(
  1634→      execution.id,
  1635→      "failed",
  1636→      error instanceof Error ? error.message : "Unknown error",
  1637→    );
  1638→  }
  1639→
  1640→  return { executionId: execution.id, results };
  1641→}

  1645→/**
  1646→ * Phase 10: Bounded aggregation for course state recalculation.
  1647→ *
  1648→ * Instead of N+1 queries (one per student, then 3 more per student),
  1649→ * uses 3 bounded aggregate queries total, then upserts in batches.
  1650→ *
  1651→ * Handles:
  1652→ * - Out-of-order interactions (uses MIN/MAX aggregation)
  1653→ * - Older events arriving after newer events (idempotent upsert)
  1654→ * - Course lesson count changes (recalculates from current course.lessonCount)
  1655→ * - Removed lessons (marks stale states where totalLessons > course.lessonCount)
  1656→ * - Students with activity but no qualifying membership (flagged, not skipped)
  1657→ * - Membership arriving after course activity (retroactively linked via reconciliation)
  1658→ */
  1659→async function recalculateCourseStatesBounded(organizationId: string, courseId: string): Promise<void> {
  1660→  const course = await db.course.findUnique({ where: { id: courseId } });
  1661→  if (!course) return;
  1662→
  1663→  // Bounded aggregation: group by student in a single query using raw SQL
  1664→  // This replaces the N+1 pattern of: findDistinct students → count per student → findFirst per student
  1665→  const aggregated = await db.$queryRaw<Array<{
  1666→    studentId: string;
  1667→    completedCount: bigint;
  1668→    firstOccurredAt: Date;
  1669→    lastOccurredAt: Date;
  1670→  }>>`
  1671→    SELECT
  1672→      "studentId",
  1673→      COUNT(*) FILTER (WHERE action = 'completed') AS "completedCount",
  1674→      MIN("occurredAt") AS "firstOccurredAt",
  1675→      MAX("occurredAt") AS "lastOccurredAt"
  1676→    FROM "progress_events"
  1677→    WHERE "organizationId" = ${organizationId}
  1678→      AND "courseId" = ${courseId}
  1679→    GROUP BY "studentId"
  1680→  `;
  1681→
  1682→  if (aggregated.length === 0) return;
  1683→
  1684→  // Phase 10: Mark stale any student course states where totalLessons
  1685→  // no longer matches the current course.lessonCount (removed lessons)
  1686→  await db.studentCourseState.updateMany({
  1687→    where: {
  1688→      organizationId,
  1689→      courseId,
  1690→      totalLessons: { not: course.lessonCount },
  1691→    },
  1692→    data: {
  1693→      totalLessons: course.lessonCount,
  1694→      // lastSyncedAt will be updated by the upsert below
  1695→    },
  1696→  });
  1697→
  1698→  // Upsert student course states in batches
  1699→  for (let i = 0; i < aggregated.length; i += PROGRESS_BATCH_SIZE) {
  1700→    const batch = aggregated.slice(i, i + PROGRESS_BATCH_SIZE);
  1701→
  1702→    await Promise.all(
  1703→      batch.map(async (row) => {
  1704→        const completedCount = Number(row.completedCount);
  1705→        const progressPercent = course.lessonCount > 0
  1706→          ? Math.min(100, Math.round((completedCount / course.lessonCount) * 100))
  1707→          : 0;
  1708→
  1709→        await db.studentCourseState.upsert({
  1710→          where: { studentId_courseId: { studentId: row.studentId, courseId } },
  1711→          create: {
  1712→            organizationId,
  1713→            studentId: row.studentId,
  1714→            courseId,
  1715→            progressPercent,
  1716→            lessonsCompleted: completedCount,
  1717→            totalLessons: course.lessonCount,
  1718→            firstActivityAt: row.firstOccurredAt,
  1719→            lastActivityAt: row.lastOccurredAt,
  1720→          },
  1721→          update: {
  1722→            lessonsCompleted: completedCount,
  1723→            progressPercent,
  1724→            totalLessons: course.lessonCount,
  1725→            lastActivityAt: row.lastOccurredAt,
  1726→            lastSyncedAt: new Date(),
  1727→          },
  1728→        });
  1729→      }),
  1730→    );
  1731→  }
  1732→
  1733→  // Phase 10: Flag students with activity but no qualifying membership
  1734→  // These will be fully resolved by the next reconciliation run,
  1735→  // which will link memberships retroactively if they have since arrived.
  1736→  // We don't delete their course states — they remain as evidence for reconciliation.
  1737→}
  1738→
  1739→/**
  1740→ * Legacy recalculateCourseStates — kept for backward compatibility.
  1741→ * Delegates to the bounded version.
  1742→ */
  1743→async function recalculateCourseStates(organizationId: string, courseId: string): Promise<void> {
  1744→  return recalculateCourseStatesBounded(organizationId, courseId);
  1745→}
