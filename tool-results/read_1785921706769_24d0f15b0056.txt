    67→    67→// ─── Bounded concurrency helper ──────────────────────────────
    68→    68→
    69→    69→/** Run items with at most `concurrency` operations in flight. */
    70→    70→async function boundedMap<T, R>(
    71→    71→  items: T[],
    72→    72→  fn: (item: T) => Promise<R>,
    73→    73→  concurrency: number = MAX_CONCURRENT_UPDATES,
    74→    74→): Promise<R[]> {
    75→    75→  const results: R[] = [];
    76→    76→  const executing: Promise<void>[] = [];
    77→    77→
    78→    78→  for (const item of items) {
    79→    79→    const p = fn(item).then((r) => { results.push(r); });
    80→    80→    executing.push(p);
    81→    81→
    82→    82→    if (executing.length >= concurrency) {
    83→    83→      await Promise.race(executing);
    84→    84→      // Remove settled promises
    85→    85→      for (let i = executing.length - 1; i >= 0; i--) {
    86→    86→        const settled = await Promise.race([
    87→    87→          executing[i].then(() => true, () => true),
    88→    88→          Promise.resolve(false),
    89→    89→        ]);
    90→    90→        if (settled) executing.splice(i, 1);
    91→    91→      }
    92→    92→    }
    93→    93→  }
    94→    94→
    95→    95→  await Promise.allSettled(executing);
    96→    96→  return results;
    97→    97→}

   101→   101→export async function syncCourses(params: SyncExecutionParams): Promise<SyncResult> {
   102→   102→  const result: SyncResult = {
   103→   103→    resource: "courses",
   104→   104→    recordsRead: 0,
   105→   105→    recordsCreated: 0,
   106→   106→    recordsUpdated: 0,
   107→   107→    recordsSkipped: 0,
   108→   108→    errors: [],
   109→   109→    warnings: [],
   110→   110→    cursor: null,
   111→   111→  };
   112→   112→
   113→   113→  // Resume from checkpoint if no explicit cursor
   114→   114→  let cursor = params.cursor ?? null;
   115→   115→  if (!cursor) {
   116→   116→    const checkpoint = await getLatestCheckpoint(params.organizationId, "courses");
   117→   117→    if (checkpoint?.cursor) {
   118→   118→      cursor = checkpoint.cursor;
   119→   119→    }
   120→   120→  }
   121→   121→
   122→   122→  let pagesProcessed = 0;
   123→   123→
   124→   124→  do {
   125→   125→    const page = await params.providers.courses.list({
   126→   126→      companyId: params.companyId,
   127→   127→      cursor,
   128→   128→      pageSize: PAGE_SIZE,
   129→   129→    });
   130→   130→
   131→   131→    result.recordsRead += page.items.length;
   132→   132→    pagesProcessed++;
   133→   133→
   134→   134→    // 1. Validate the full page
   135→   135→    const validItems = page.items.filter((course) => {
   136→   136→      if (!course.id) {
   137→   137→        result.errors.push(`Course missing id — skipped`);
   138→   138→        result.recordsSkipped++;
   139→   139→        return false;
   140→   140→      }
   141→   141→      return true;
   142→   142→    });
   143→   143→
   144→   144→    if (validItems.length === 0) {
   145→   145→      cursor = page.nextCursor;
   146→   146→      continue;
   147→   147→    }
   148→   148→
   149→   149→    // 2. Load existing external IDs in one bounded query
   150→   150→    const externalIds = validItems.map((c) => c.id);
   151→   151→    const existing = await db.course.findMany({
   152→   152→      where: {
   153→   153→        organizationId: params.organizationId,
   154→   154→        externalCourseId: { in: externalIds },
   155→   155→      },
   156→   156→      select: { id: true, externalCourseId: true },
   157→   157→    });
   158→   158→    const existingMap = new Map(existing.map((e) => [e.externalCourseId, e.id]));
   159→   159→
   160→   160→    // 3. Split into creates and updates
   161→   161→    const toCreate = validItems.filter((c) => !existingMap.has(c.id));
   162→   162→    const toUpdate = validItems.filter((c) => existingMap.has(c.id));
   163→   163→
   164→   164→    // 4. Batch creates with skipDuplicates
   165→   165→    if (toCreate.length > 0) {
   166→   166→      try {
   167→   167→        await db.course.createMany({
   168→   168→          data: toCreate.map((course) => ({
   169→   169→            organizationId: params.organizationId,
   170→   170→            externalCourseId: course.id,
   171→   171→            externalExperienceId: course.experienceId,
   172→   172→            name: course.title ?? "Untitled Course",
   173→   173→            lessonCount: course.lessonCount,
   174→   174→          })),
   175→   175→          skipDuplicates: true,
   176→   176→        });
   177→   177→        result.recordsCreated += toCreate.length;
   178→   178→      } catch (error) {
   179→   179→        result.errors.push(`Course batch create: ${error instanceof Error ? error.message : "unknown"}`);
   180→   180→        result.recordsSkipped += toCreate.length;
   181→   181→      }
   182→   182→    }
   183→   183→
   184→   184→    // 5. Batch updates with bounded concurrency
   185→   185→    if (toUpdate.length > 0) {
   186→   186→      const updateResults = await boundedMap(toUpdate, async (course) => {
   187→   187→        const existingId = existingMap.get(course.id)!;
   188→   188→        try {
   189→   189→          await db.course.update({
   190→   190→            where: { id: existingId },
   191→   191→            data: {
   192→   192→              name: course.title ?? undefined,
   193→   193→              lessonCount: course.lessonCount || undefined,
   194→   194→              externalExperienceId: course.experienceId || undefined,
   195→   195→            },
   196→   196→          });
   197→   197→          return true;
   198→   198→        } catch (error) {
   199→   199→          result.errors.push(`Course ${course.id}: ${error instanceof Error ? error.message : "unknown"}`);
   200→   200→          return false;
   201→   201→        }
   202→   202→      });
   203→   203→      result.recordsUpdated += updateResults.filter(Boolean).length;
   204→   204→      result.recordsSkipped += updateResults.filter((r) => !r).length;
   205→   205→    }
   206→   206→
   207→   207→    // 6. Commit page — persist checkpoint
   208→   208→    cursor = page.nextCursor;
   209→   209→    if (params.syncExecutionId) {
   210→   210→      await persistCheckpoint(
   211→   211→        params.organizationId,
   212→   212→        "courses",
   213→   213→        cursor,
   214→   214→        null,
   215→   215→        params.syncExecutionId,
   216→   216→        pagesProcessed,
   217→   217→      );
   218→   218→    }
   219→   219→  } while (cursor);
   220→   220→
   221→   221→  result.cursor = cursor;
   222→   222→
   223→   223→  // 7. Emit usage and audit data
   224→   224→  await recordAuditEvent({
   225→   225→    organizationId: params.organizationId,
   226→   226→    action: "synced",
   227→   227→    objectType: "course",
   228→   228→    objectId: "bulk",
   229→   229→    reason: `Synced ${result.recordsRead} courses`,
   230→   230→    metadata: {
   231→   231→      created: result.recordsCreated,
   232→   232→      updated: result.recordsUpdated,
   233→   233→      skipped: result.recordsSkipped,
   234→   234→      errors: result.errors.length,
   235→   235→    },
   236→   236→  });
   237→   237→
   238→   238→  return result;
   239→   239→}

   479→   479→              status: safeStatus,

   481→   481→              renewalDate: membership.renewalDate ? new Date(membership.renewalDate) : null,
   482→   482→              cancelledAt: membership.cancelledAt ? new Date(membership.cancelledAt) : null,

   492→   492→        result.recordsSkipped++;
   493→   493→      }
   494→   494→    }
   495→   495→
   496→   496→    // Persist checkpoint
   497→   497→    cursor = page.nextCursor;
   498→   498→    if (params.syncExecutionId) {
   499→   499→      await persistCheckpoint(
   500→   500→        params.organizationId,
   501→   501→        "memberships",
   502→   502→        cursor,
   503→   503→        null,
   504→   504→        params.syncExecutionId,
   505→   505→        pagesProcessed,
   506→   506→      );
   507→   507→    }
   508→   508→  } while (cursor);
   509→   509→
   510→   510→  result.cursor = cursor;
   511→   511→
   512→   512→  await recordAuditEvent({
   513→   513→    organizationId: params.organizationId,
   514→   514→    action: "synced",
   515→   515→    objectType: "membership",
   516→   516→    objectId: "bulk",
   517→   517→    reason: `Synced ${result.recordsRead} memberships`,
   518→   518→    metadata: {
   519→   519→      created: result.recordsCreated,
   520→   520→      updated: result.recordsUpdated,
   521→   521→      skipped: result.recordsSkipped,
   522→   522→      warnings: result.warnings.length,
   523→   523→    },
   524→   524→  });
   525→   525→
   526→   526→  return result;
   527→   527→}

   534→   534→  const result: SyncResult = {
   535→   535→    resource: "progress",
   536→   536→    recordsRead: 0,
   537→   537→    recordsCreated: 0,
   538→   538→    recordsUpdated: 0,
   539→   539→    recordsSkipped: 0,
   540→   540→    errors: [],
   541→   541→    warnings: [],
   542→   542→    cursor: null,
   543→   543→  };
   544→   544→
   545→   545→  // Find the internal course
   546→   546→  const course = await db.course.findFirst({
   547→   547→    where: {
   548→   548→      organizationId: params.organizationId,
   549→   549→      id: params.courseId,
   550→   550→    },
   551→   551→  });
   552→   552→
   553→   553→  if (!course) {
   554→   554→    result.errors.push(`Course ${params.courseId} not found`);
   555→   555→    return result;
   556→   556→  }
   557→   557→
   558→   558→  let cursor = params.cursor ?? null;
   559→   559→  if (!cursor) {
   560→   560→    const checkpoint = await getLatestCheckpoint(params.organizationId, "progress");
   561→   561→    if (checkpoint?.cursor) {
   562→   562→      cursor = checkpoint.cursor;
   563→   563→    }
   564→   564→  }
   565→   565→
   566→   566→  let pagesProcessed = 0;
   567→   567→
   568→   568→  do {
   569→   569→    const page = await params.providers.progress.listLessonInteractions({
   570→   570→      companyId: params.companyId,
   571→   571→      courseId: course.externalCourseId,
   572→   572→      cursor,
   573→   573→      pageSize: PAGE_SIZE,
   574→   574→    });
   575→   575→
   576→   576→    result.recordsRead += page.items.length;
   577→   577→    pagesProcessed++;
   578→   578→
   579→   579→    for (const interaction of page.items) {
   580→   580→      try {
   581→   581→        // Find the student
   582→   582→        const student = await db.student.findFirst({
   583→   583→          where: {
   584→   584→            organizationId: params.organizationId,
   585→   585→            whopUserId: interaction.userId,
   586→   586→          },
   587→   587→        });
   588→   588→
   589→   589→        if (!student) {
   590→   590→          result.recordsSkipped++;
   591→   591→          continue;
   592→   592→        }
   593→   593→
   594→   594→        // Use the external interaction ID for idempotency
   595→   595→        // The unique constraint prevents duplicate counting
   596→   596→        try {
   597→   597→          await db.progressEvent.create({
   598→   598→            data: {
   599→   599→              organizationId: params.organizationId,
   600→   600→              studentId: student.id,
   601→   601→              courseId: course.id,
   602→   602→              externalInteractionId: interaction.id,
   603→   603→              lessonIndex: 0,
   604→   604→              lessonTitle: interaction.lessonTitle,
   605→   605→              action: interaction.completed ? "completed" : "started",
   606→   606→              occurredAt: new Date(interaction.createdAt),
   607→   607→            },
   608→   608→          });
   609→   609→          result.recordsCreated++;
   610→   610→        } catch {
   611→   611→          // Unique constraint violation — already recorded (idempotent)
   612→   612→          result.recordsSkipped++;
   613→   613→        }
   614→   614→      } catch (error) {
   615→   615→        result.errors.push(`Interaction ${interaction.id}: ${error instanceof Error ? error.message : "unknown"}`);
   616→   616→        result.recordsSkipped++;
   617→   617→      }
   618→   618→    }
   619→   619→
   620→   620→    // Persist checkpoint
   621→   621→    cursor = page.nextCursor;
   622→   622→    if (params.syncExecutionId) {
   623→   623→      await persistCheckpoint(
   624→   624→        params.organizationId,
   625→   625→        "progress",
   626→   626→        cursor,
   627→   627→        null,
   628→   628→        params.syncExecutionId,
   629→   629→        pagesProcessed,
   630→   630→      );
   631→   631→    }
   632→   632→  } while (cursor);
   633→   633→
   634→   634→  // Recalculate course states from the interaction table
   635→   635→  await recalculateCourseStates(params.organizationId, course.id);
   636→   636→
   637→   637→  result.cursor = cursor;
   638→   638→
   639→   639→  await recordAuditEvent({
   640→   640→    organizationId: params.organizationId,
   641→   641→    action: "synced",
   642→   642→    objectType: "progress",
   643→   643→    objectId: course.id,
   644→   644→    reason: `Synced ${result.recordsRead} progress events`,
   645→   645→    metadata: {
   646→   646→      created: result.recordsCreated,
   647→   647→      skipped: result.recordsSkipped,
   648→   648→    },
   649→   649→  });
   650→   650→
   651→   651→  return result;
   652→   652→}

   656→   656→export interface ReconciliationResult {
   657→   657→  matched: number;
   658→   658→  membershipWithoutCourseActivity: number;
   659→   659→  courseActivityWithoutMembership: number;
   660→   660→  unmappedProduct: number;
   661→   661→  missingSourceFields: number;
   662→   662→  staleSourceRecord: number;
   663→   663→}
   664→   664→
   665→   665→export async function reconcile(
   666→   666→  organizationId: string,
   667→   667→  courseId: string,
   668→   668→  syncExecutionId?: string,
   669→   669→): Promise<ReconciliationResult> {
   670→   670→  const result: ReconciliationResult = {
   671→   671→    matched: 0,
   672→   672→    membershipWithoutCourseActivity: 0,
   673→   673→    courseActivityWithoutMembership: 0,
   674→   674→    unmappedProduct: 0,
   675→   675→    missingSourceFields: 0,
   676→   676→    staleSourceRecord: 0,
   677→   677→  };
   678→   678→
   679→   679→  // Create a reconciliation run record
   680→   680→  const run = await createReconciliationRun(organizationId, courseId, syncExecutionId);
   681→   681→
   682→   682→  // Get all memberships for this organization that map to this course
   683→   683→  const mappings = await db.productCourseMapping.findMany({
   684→   684→    where: { organizationId, courseId, isConfirmed: true },
   685→   685→    include: { product: { include: { memberships: true } } },
   686→   686→  });
   687→   687→
   688→   688→  const mappedProductIds = new Set(mappings.map((m) => m.productId));
   689→   689→
   690→   690→  // Get all students with course activity
   691→   691→  const courseStates = await db.studentCourseState.findMany({
   692→   692→    where: { organizationId, courseId },
   693→   693→  });
   694→   694→
   695→   695→  const studentsWithActivity = new Set(courseStates.map((s) => s.studentId));
   696→   696→
   697→   697→  // Check each membership
   698→   698→  for (const mapping of mappings) {
   699→   699→    for (const membership of mapping.product.memberships) {
   700→   700→      if (studentsWithActivity.has(membership.studentId)) {
   701→   701→        result.matched++;
   702→   702→      } else {
   703→   703→        result.membershipWithoutCourseActivity++;
   704→   704→      }
   705→   705→    }
   706→   706→  }
   707→   707→
   708→   708→  // Check for course activity without a mapped membership
   709→   709→  for (const state of courseStates) {
   710→   710→    const hasMembership = await db.membership.findFirst({
   711→   711→      where: {
   712→   712→        organizationId,
   713→   713→        studentId: state.studentId,
   714→   714→        productId: { in: Array.from(mappedProductIds) },
   715→   715→      },
   716→   716→    });
   717→   717→
   718→   718→    if (!hasMembership) {
   719→   719→      result.courseActivityWithoutMembership++;
   720→   720→    }
   721→   721→  }
   722→   722→
   723→   723→  // Complete the reconciliation run record
   724→   724→  await completeReconciliationRun(run.id, result);
   725→   725→
   726→   726→  return result;
   727→   727→}

   731→   731→export interface CandidateDetectionResult {
   732→   732→  candidatesFound: number;
   733→   733→  snapshotsCreated: number;
   734→   734→  errors: string[];
   735→   735→}
   736→   736→
   737→   737→export async function detectCandidates(
   738→   738→  organizationId: string,
   739→   739→  campaignId: string,
   740→   740→): Promise<CandidateDetectionResult> {
   741→   741→  const result: CandidateDetectionResult = {
   742→   742→    candidatesFound: 0,
   743→   743→    snapshotsCreated: 0,
   744→   744→    errors: [],
   745→   745→  };
   746→   746→
   747→   747→  const campaign = await db.campaign.findFirst({
   748→   748→    where: { id: campaignId, organizationId, type: "activation_rescue" },
   749→   749→    include: { organization: true, versions: { orderBy: { versionNumber: "desc" }, take: 1 } },
   750→   750→  });
   751→   751→
   752→   752→  if (!campaign) {
   753→   753→    result.errors.push("Campaign not found or not an Activation Rescue campaign");
   754→   754→    return result;
   755→   755→  }
   756→   756→
   757→   757→  const latestVersion = campaign.versions[0];
   758→   758→
   759→   759→  // Find all confirmed product-course mappings for this org
   760→   760→  const mappings = await db.productCourseMapping.findMany({
   761→   761→    where: { organizationId, isConfirmed: true },
   762→   762→    include: {
   763→   763→      product: { include: { memberships: { where: { status: { in: ["active", "trialing"] } } } } },
   764→   764→      course: true,
   765→   765→    },
   766→   766→  });
   767→   767→
   768→   768→  const now = new Date();
   769→   769→
   770→   770→  for (const mapping of mappings) {
   771→   771→    for (const membership of mapping.product.memberships) {
   772→   772→      // Check if the student has any course activity
   773→   773→      const hasActivity = await db.studentCourseState.findFirst({
   774→   774→        where: {
   775→   775→          organizationId,
   776→   776→          studentId: membership.studentId,
   777→   777→          courseId: mapping.courseId,
   778→   778→        },
   779→   779→      });
   780→   780→
   781→   781→      if (hasActivity && hasActivity.lessonsCompleted > 0) {
   782→   782→        continue; // Has course activity — not a candidate
   783→   783→      }
   784→   784→
   785→   785→      // Check activation delay
   786→   786→      const daysSinceJoin = (now.getTime() - membership.joinedAt.getTime()) / (1000 * 60 * 60 * 24);
   787→   787→      if (daysSinceJoin < mapping.activationDelayDays) {
   788→   788→        continue; // Too soon
   789→   789→      }
   790→   790→
   791→   791→      // Check suppression
   792→   792→      const suppressed = await db.suppression.findFirst({
   793→   793→        where: { organizationId, studentId: membership.studentId },
   794→   794→      });
   795→   795→      if (suppressed) continue;
   796→   796→
   797→   797→      // Check existing active intervention
   798→   798→      const existingIntervention = await db.intervention.findFirst({
   799→   799→        where: {
   800→   800→          organizationId,
   801→   801→          studentId: membership.studentId,
   802→   802→          campaignId,
   803→   803→          state: { notIn: ["dismissed", "stopped", "failed"] },
   804→   804→        },
   805→   805→      });
   806→   806→      if (existingIntervention) continue;
   807→   807→
   808→   808→      // Check cooldown
   809→   809→      const lastIntervention = await db.intervention.findFirst({
   810→   810→        where: { organizationId, studentId: membership.studentId, campaignId },
   811→   811→        orderBy: { createdAt: "desc" },
   812→   812→      });
   813→   813→
   814→   814→      if (lastIntervention) {
   815→   815→        const daysSinceLast = (now.getTime() - lastIntervention.createdAt.getTime()) / (1000 * 60 * 60 * 24);
   816→   816→        if (daysSinceLast < campaign.cooldownDays) continue;
   817→   817→      }
   818→   818→
   819→   819→      // Check max messages
   820→   820→      const recentCount = await db.intervention.count({
   821→   821→        where: {
   822→   822→          organizationId,
   823→   823→          studentId: membership.studentId,
   824→   824→          campaignId,
   825→   825→          createdAt: { gte: new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000) },
   826→   826→          state: { in: ["notification_accepted", "delivered"] },
   827→   827→        },
   828→   828→      });
   829→   829→      if (recentCount >= campaign.maxMessagesPerStudent) continue;
   830→   830→
   831→   831→      // Candidate found — create eligibility snapshot
   832→   832→      result.candidatesFound++;
   833→   833→
   834→   834→      const evidence = {
   835→   835→        detectedAt: now.toISOString(),
   836→   836→        membershipId: membership.id,
   837→   837→        membershipStatus: membership.status,
   838→   838→        joinedAt: membership.joinedAt.toISOString(),
   839→   839→        daysSinceJoin: Math.floor(daysSinceJoin),
   840→   840→        activationDelayDays: mapping.activationDelayDays,
   841→   841→        courseId: mapping.courseId,
   842→   842→        productId: mapping.productId,
   843→   843→        campaignId,
   844→   844→        campaignVersionId: latestVersion?.id,
   845→   845→      };
   846→   846→
   847→   847→      try {
   848→   848→        await db.eligibilitySnapshot.create({
   849→   849→          data: {
   850→   850→            organizationId,
   851→   851→            studentId: membership.studentId,
   852→   852→            campaignId,
   853→   853→            campaignVersionId: latestVersion?.id,
   854→   854→            state: "eligible",
   855→   855→            evidenceJson: evidence as any,
   856→   856→            detectedAt: now,
   857→   857→            expiresAt: new Date(now.getTime() + 7 * 24 * 60 * 60 * 1000),
   858→   858→          },
   859→   859→        });
   860→   860→        result.snapshotsCreated++;
   861→   861→      } catch (error) {
   862→   862→        result.errors.push(
   863→   863→          `Student ${membership.studentId}: ${error instanceof Error ? error.message : "unknown"}`,
   864→   864→        );
   865→   865→      }
   866→   866→    }
   867→   867→  }
   868→   868→
   869→   869→  return result;
   870→   870→}

   872→   872→// ─── Orchestrated sync execution ─────────────────────────────
   873→   873→
   874→   874→/**
   875→   875→ * Run a full sync with durable execution records and stage tracking.
   876→   876→ * Returns the sync execution ID for observability.
   877→   877→ */
   878→   878→export async function runFullSync(
   879→   879→  params: SyncExecutionParams & { resources?: string[] },
   880→   880→): Promise<{ executionId: string; results: SyncResult[] }> {
   881→   881→  const resources = params.resources ?? ["courses", "products", "memberships"];
   882→   882→  const results: SyncResult[] = [];
   883→   883→
   884→   884→  // Create the execution record
   885→   885→  const execution = await createSyncExecution({
   886→   886→    orgId: params.organizationId,
   887→   887→    trigger: params.requestedBy ? "manual" : "scheduled",
   888→   888→    requestedBy: params.requestedBy,
   889→   889→    jobId: params.syncExecutionId,
   890→   890→  });
   891→   891→
   892→   892→  try {
   893→   893→    for (const resource of resources) {
   894→   894→      const stage = await createSyncStage(execution.id, resource);
   895→   895→
   896→   896→      let syncResult: SyncResult;
   897→   897→      switch (resource) {
   898→   898→        case "courses":
   899→   899→          syncResult = await syncCourses({ ...params, syncExecutionId: execution.id });
   900→   900→          break;
   901→   901→        case "products":
   902→   902→          syncResult = await syncProducts({ ...params, syncExecutionId: execution.id });
   903→   903→          break;
   904→   904→        case "memberships":
   905→   905→          syncResult = await syncMemberships({ ...params, syncExecutionId: execution.id });
   906→   906→          break;
   907→   907→        default:
   908→   908→          await completeSyncStage(stage.id, {
   909→   909→            pagesProcessed: 0, recordsRead: 0, recordsCreated: 0,
   910→   910→            recordsUpdated: 0, recordsSkipped: 0,
   911→   911→          }, "skipped");
   912→   912→          continue;
   913→   913→      }
   914→   914→
   915→   915→      results.push(syncResult);
   916→   916→
   917→   917→      // Complete the stage with counts
   918→   918→      await completeSyncStage(stage.id, {
   919→   919→        pagesProcessed: 0, // Pages are tracked inside the sync functions
   920→   920→        recordsRead: syncResult.recordsRead,
   921→   921→        recordsCreated: syncResult.recordsCreated,
   922→   922→        recordsUpdated: syncResult.recordsUpdated,
   923→   923→        recordsSkipped: syncResult.recordsSkipped,
   924→   924→      });
   925→   925→    }
   926→   926→
   927→   927→    await completeSyncExecution(execution.id, "completed");
   928→   928→  } catch (error) {
   929→   929→    await completeSyncExecution(
   930→   930→      execution.id,
   931→   931→      "failed",
   932→   932→      error instanceof Error ? error.message : "Unknown error",
   933→   933→    );
   934→   934→  }
   935→   935→
   936→   936→  return { executionId: execution.id, results };
   937→   937→}

   941→   941→async function recalculateCourseStates(organizationId: string, courseId: string): Promise<void> {
   942→   942→  const course = await db.course.findUnique({ where: { id: courseId } });
   943→   943→  if (!course) return;
   944→   944→
   945→   945→  // Get all students with progress events for this course
   946→   946→  const students = await db.progressEvent.findMany({
   947→   947→    where: { organizationId, courseId },
   948→   948→    select: { studentId: true },
   949→   949→    distinct: ["studentId"],
   950→   950→  });
   951→   951→
   952→   952→  for (const { studentId } of students) {
   953→   953→    const completedCount = await db.progressEvent.count({
   954→   954→      where: {
   955→   955→        organizationId,
   956→   956→        studentId,
   957→   957→        courseId,
   958→   958→        action: "completed",
   959→   959→      },
   960→   960→    });
   961→   961→
   962→   962→    const firstEvent = await db.progressEvent.findFirst({
   963→   963→      where: { organizationId, studentId, courseId },
   964→   964→      orderBy: { occurredAt: "asc" },
   965→   965→    });
   966→   966→
   967→   967→    const lastEvent = await db.progressEvent.findFirst({
   968→   968→      where: { organizationId, studentId, courseId },
   969→   969→      orderBy: { occurredAt: "desc" },
   970→   970→    });
   971→   971→
   972→   972→    const progressPercent = course.lessonCount > 0
   973→   973→      ? Math.round((completedCount / course.lessonCount) * 100)
   974→   974→      : 0;
   975→   975→
   976→   976→    await db.studentCourseState.upsert({
   977→   977→      where: { studentId_courseId: { studentId, courseId } },
   978→   978→      create: {
   979→   979→        organizationId,
   980→   980→        studentId,
   981→   981→        courseId,
   982→   982→        progressPercent,
   983→   983→        lessonsCompleted: completedCount,
   984→   984→        totalLessons: course.lessonCount,
   985→   985→        firstActivityAt: firstEvent?.occurredAt ?? new Date(),
   986→   986→        lastActivityAt: lastEvent?.occurredAt ?? new Date(),
   987→   987→      },
   988→   988→      update: {
   989→   989→        lessonsCompleted: completedCount,
   990→   990→        progressPercent,
   991→   991→        lastActivityAt: lastEvent?.occurredAt ?? undefined,
   992→   992→      },
   993→   993→    });
   994→   994→  }
   995→   995→}
