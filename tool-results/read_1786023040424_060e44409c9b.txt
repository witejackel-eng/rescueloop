     5→// avoid calling getInngestClient() at module import time, which would
     6→// crash `next build` when Inngest is not configured.

    27→// ─── Job function definitions (lazy) ─────────────────────────
    28→
    29→/**
    30→ * Create and return the Inngest job functions.
    31→ * Call this only when Inngest is configured — it calls getInngestClient().
    32→ */
    33→export function getJobFunctions(): any[] {
    34→  if (!isInngestReady()) return [];
    35→  const client = getInngestClient();
    36→
    37→  const processWebhook = client.createFunction(
    38→  {
    39→    id: "process-webhook",
    40→    retries: 5,
    41→  },
    42→  { event: EVENTS.webhookReceived },
    43→  async ({ event, step }) => {
    44→    const { receiptId } = event.data as { receiptId: string };
    45→
    46→    // Step 1: Mark as processing (idempotent — skip if already processed)
    47→    await step.run("mark-processing", async () => {
    48→      const receipt = await db.webhookReceipt.findUnique({
    49→        where: { id: receiptId },
    50→      });
    51→      if (!receipt || receipt.status === "processed") {
    52→        return { skipped: true };
    53→      }
    54→      await db.webhookReceipt.update({
    55→        where: { id: receiptId },
    56→        data: { status: "processing", attemptCount: { increment: 1 } },
    57→      });
    58→      return { skipped: false };
    59→    });
    60→
    61→    const receipt = await db.webhookReceipt.findUnique({
    62→      where: { id: receiptId },
    63→    });
    64→
    65→    if (!receipt || receipt.status === "processed") {
    66→      return { processed: false, reason: "already-processed" };
    67→    }
    68→
    69→    // The payload is the verified event object from whopsdk.webhooks.unwrap()
    70→    const eventPayload = receipt.payloadJson as Record<string, unknown>;
    71→
    72→    try {
    73→      // Step 2: Process based on event type (using official Whop event names)
    74→      if (receipt.eventType === "membership.activated") {
    75→        await step.run("handle-membership-activated", async () => {
    76→          await handleMembershipActivated(receipt.organizationId, eventPayload as unknown as WhopMembershipEvent, receiptId);
    77→        });
    78→      } else if (receipt.eventType === "membership.deactivated") {
    79→        await step.run("handle-membership-deactivated", async () => {
    80→          await handleMembershipDeactivated(receipt.organizationId, eventPayload as unknown as WhopMembershipEvent, receiptId);
    81→        });
    82→      } else if (receipt.eventType === "payment.succeeded") {
    83→        await step.run("handle-payment-succeeded", async () => {
    84→          await handlePaymentSucceeded(receipt.organizationId, eventPayload as unknown as WhopPaymentEvent, receiptId);
    85→        });
    86→      } else if (receipt.eventType === "course_lesson_interaction.completed") {
    87→        await step.run("handle-lesson-completed", async () => {
    88→          await handleLessonCompleted(receipt.organizationId, eventPayload as unknown as WhopLessonInteractionEvent, receiptId);
    89→        });
    90→      } else {
    91→        // Unknown event type — acknowledge but don't process
    92→        await step.run("mark-unknown", async () => {
    93→          await db.webhookReceipt.update({
    94→            where: { id: receiptId },
    95→            data: {
    96→              status: "processed",
    97→              processedAt: new Date(),
    98→              lastError: `Unknown event type: ${receipt.eventType}`,
    99→            },
   100→          });
   101→        });
   102→        return { processed: false, reason: "unknown-event-type" };
   103→      }
   104→
   105→      // Step 3: Mark as processed
   106→      await step.run("mark-processed", async () => {
   107→        await db.webhookReceipt.update({
   108→          where: { id: receiptId },
   109→          data: { status: "processed", processedAt: new Date() },
   110→        });
   111→      });
   112→
   113→      return { processed: true };
   114→    } catch (error) {
   115→      // Record the failure
   116→      await db.webhookReceipt.update({
   117→        where: { id: receiptId },
   118→        data: {
   119→          status: "failed",
   120→          lastError: error instanceof Error ? error.message : "Unknown error",
   121→        },
   122→      });
   123→      throw error; // Let Inngest retry
   124→    }
   125→  },
   126→);
   127→
   128→/**
   129→ * Intervention delivery job.
   130→ *
   131→ * Before every send, transactionally re-checks ALL safety conditions.
   132→ * Uses a stable idempotency key derived from the intervention + delivery
   133→ * version (never Date.now()).
   134→ *
   135→ * Calls the official notification API:
   136→ *   whopsdk.notifications.create({ experience_id, title, content, user_ids, rest_path })
   137→ *
   138→ * Records the delivery state truthfully:
   139→ * - api_accepted: the Whop API returned { success: true }
   140→ * - failed: the API threw or returned { success: false }
   141→ * - stopped_before_send: a safety check failed
   142→ *
   143→ * Does NOT set the intervention to "delivered" merely because the API
   144→ * accepted the notification. Uses "notification_accepted" as the
   145→ * intervention state.
   146→ */
   147→  const deliverIntervention = client.createFunction(
   148→  {
   149→    id: "deliver-intervention",
   150→    retries: 3,
   151→  },
   152→  { event: EVENTS.deliverIntervention },
   153→  async ({ event, step }) => {
   154→    const { interventionId } = event.data as { interventionId: string };
   155→
   156→    const intervention = await db.intervention.findUnique({
   157→      where: { id: interventionId },
   158→      include: {
   159→        student: { include: { memberships: { include: { product: { include: { mappings: true } } } }, suppressions: true } },
   160→        campaign: true,
   161→        organization: true,
   162→      },
   163→    });
   164→
   165→    if (!intervention) {
   166→      return { delivered: false, reason: "not-found" };
   167→    }
   168→
   169→    // Stable idempotency key — derived from intervention + campaign version, NOT Date.now()
   170→    const idempotencyKey = `delivery-${intervention.id}-${intervention.campaignVersionId ?? "v1"}`;
   171→
   172→    // Check for existing accepted delivery (idempotency)
   173→    const existingAttempt = await db.deliveryAttempt.findUnique({
   174→      where: { idempotencyKey },
   175→    });
   176→
   177→    if (existingAttempt?.state === "api_accepted") {
   178→      return { delivered: false, reason: "already-accepted" };
   179→    }
   180→
   181→    // ─── Transactional safety re-checks ───────────────────────
   182→    const safetyResult = await step.run("safety-recheck", async () => {
   183→      return await db.$transaction(async (tx) => {
   184→        const checks: { condition: string; passed: boolean; detail: string }[] = [];
   185→
   186→        // 1. Organisation status is active
   187→        const orgActive = intervention.organization.status === "active";
   188→        checks.push({ condition: "org_status_active", passed: orgActive, detail: orgActive ? "OK" : `Status: ${intervention.organization.status}` });
   189→
   190→        // 2. Organisation global pause is false
   191→        const notPaused = !intervention.organization.isPaused;
   192→        checks.push({ condition: "org_not_paused", passed: notPaused, detail: notPaused ? "OK" : "Organisation is paused" });
   193→
   194→        // 3. Campaign is active
   195→        const campaignActive = intervention.campaign.status === "active";
   196→        checks.push({ condition: "campaign_active", passed: campaignActive, detail: campaignActive ? "OK" : `Campaign: ${intervention.campaign.status}` });
   197→
   198→        // 4. Campaign uses manual approval
   199→        const manualApproval = intervention.campaign.approvalMode === "manual";
   200→        checks.push({ condition: "manual_approval", passed: manualApproval, detail: manualApproval ? "OK" : "Automatic mode not supported in pilot" });
   201→
   202→        // 5. Intervention is approved or validly scheduled
   203→        const approved = intervention.state === "approved" || intervention.state === "scheduled";
   204→        checks.push({ condition: "intervention_approved", passed: approved, detail: approved ? "OK" : `State: ${intervention.state}` });
   205→
   206→        // 6. scheduledFor <= now (if scheduled)
   207→        const now = new Date();
   208→        const scheduledOk = !intervention.scheduledFor || intervention.scheduledFor <= now;
   209→        checks.push({ condition: "schedule_reached", passed: scheduledOk, detail: scheduledOk ? "OK" : "Scheduled for future" });
   210→
   211→        // 7. Relevant membership is active or trialing
   212→        const activeMembership = intervention.student.memberships.find(
   213→          (m) => m.status === "active" || m.status === "trialing",
   214→        );
   215→        const membershipActive = !!activeMembership;
   216→        checks.push({ condition: "membership_active", passed: membershipActive, detail: membershipActive ? `Status: ${activeMembership!.status}` : "No active membership" });
   217→
   218→        // 8. Membership grants access to the mapped course
   219→        const productMapsToCourse = activeMembership?.product.mappings.some(
   220→          (m) => m.isConfirmed,
   221→        ) ?? false;
   222→        checks.push({ condition: "product_mapped", passed: productMapsToCourse, detail: productMapsToCourse ? "OK" : "Product not mapped to course" });
   223→
   224→        // 9. Membership is not cancelling or ending
   225→        const notCancelling = activeMembership?.status !== "cancelling" && activeMembership?.status !== "cancelled";
   226→        checks.push({ condition: "membership_not_ending", passed: notCancelling, detail: notCancelling ? "OK" : "Membership ending" });
   227→
   228→        // 10. No course interaction has appeared (check progress events)
   229→        const progressEvents = await tx.progressEvent.findFirst({
   230→          where: { studentId: intervention.studentId, organizationId: intervention.organizationId },
   231→        });
   232→        const noCourseActivity = !progressEvents;
   233→        checks.push({ condition: "no_course_activity", passed: noCourseActivity, detail: noCourseActivity ? "OK" : "Course activity detected — stopping" });
   234→
   235→        // 11. No student response has appeared
   236→        const existingResponse = await tx.studentResponse.findFirst({
   237→          where: { interventionId: intervention.id },
   238→        });
   239→        const noResponse = !existingResponse;
   240→        checks.push({ condition: "no_student_response", passed: noResponse, detail: noResponse ? "OK" : "Student already responded" });
   241→
   242→        // 12. No organisation/course suppression exists
   243→        const notSuppressed = intervention.student.suppressions.length === 0;
   244→        checks.push({ condition: "not_suppressed", passed: notSuppressed, detail: notSuppressed ? "OK" : "Student opted out" });
   245→
   246→        // 13. Campaign cooldown has elapsed
   247→        const cooldownDays = intervention.campaign.cooldownDays;
   248→        const lastIntervention = await tx.intervention.findFirst({
   249→          where: {
   250→            studentId: intervention.studentId,
   251→            organizationId: intervention.organizationId,
   252→            id: { not: intervention.id },
   253→          },
   254→          orderBy: { createdAt: "desc" },
   255→        });
   256→        const cooldownElapsed = !lastIntervention ||
   257→          (now.getTime() - lastIntervention.createdAt.getTime()) / (1000 * 60 * 60 * 24) >= cooldownDays;
   258→        checks.push({ condition: "cooldown_elapsed", passed: cooldownElapsed, detail: cooldownElapsed ? "OK" : `Within ${cooldownDays}-day cooldown` });
   259→
   260→        // 14. Maximum messages in the configured period has not been reached
   261→        const maxMessages = intervention.campaign.maxMessagesPerStudent;
   262→        const recentInterventions = await tx.intervention.count({
   263→          where: {
   264→            studentId: intervention.studentId,
   265→            organizationId: intervention.organizationId,
   266→            campaignId: intervention.campaignId,
   267→            createdAt: { gte: new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000) },
   268→            state: { in: ["delivered", "notification_accepted"] },
   269→          },
   270→        });
   271→        const underMaxMessages = recentInterventions < maxMessages;
   272→        checks.push({ condition: "under_max_messages", passed: underMaxMessages, detail: underMaxMessages ? `${recentInterventions}/${maxMessages}` : `Max reached: ${recentInterventions}/${maxMessages}` });
   273→
   274→        // 15. Current student-local time is outside quiet hours
   275→        // (Simplified — real implementation would use student timezone)
   276→        const quietHoursOk = checkQuietHours(intervention.campaign.quietHoursStart, intervention.campaign.quietHoursEnd, now);
   277→        checks.push({ condition: "outside_quiet_hours", passed: quietHoursOk, detail: quietHoursOk ? "OK" : "Within quiet hours" });
   278→
   279→        // 16. No equivalent active intervention exists
   280→        const equivalentActive = await tx.intervention.findFirst({
   281→          where: {
   282→            studentId: intervention.studentId,
   283→            organizationId: intervention.organizationId,
   284→            campaignId: intervention.campaignId,
   285→            id: { not: intervention.id },
   286→            state: { in: ["approved", "scheduled", "queued", "delivered", "notification_accepted"] },
   287→          },
   288→        });
   289→        const noEquivalent = !equivalentActive;
   290→        checks.push({ condition: "no_equivalent_active", passed: noEquivalent, detail: noEquivalent ? "OK" : "Equivalent intervention exists" });
   291→
   292→        // 17. Stable idempotency key has not already been accepted
   293→        const existingAccepted = await tx.deliveryAttempt.findUnique({
   294→          where: { idempotencyKey },
   295→        });
   296→        const notAlreadyAccepted = !existingAccepted || existingAccepted.state !== "api_accepted";
   297→        checks.push({ condition: "idempotency_clear", passed: notAlreadyAccepted, detail: notAlreadyAccepted ? "OK" : "Already accepted" });
   298→
   299→        const allPassed = checks.every((c) => c.passed);
   300→
   301→        // Create the delivery attempt record inside the transaction
   302→        if (!allPassed) {
   303→          await tx.deliveryAttempt.create({
   304→            data: {
   305→              interventionId: intervention.id,
   306→              state: "stopped_before_send",
   307→              idempotencyKey,
   308→              errorMessage: JSON.stringify(checks.filter((c) => !c.passed)),
   309→            },
   310→          });
   311→          await tx.intervention.update({
   312→            where: { id: intervention.id },
   313→            data: { state: "stopped" },
   314→          });
   315→        }
   316→
   317→        return { allPassed, checks, activeMembership };
   318→      });
   319→    });
   320→
   321→    if (!safetyResult.allPassed) {
   322→      await recordAuditEvent({
   323→        organizationId: intervention.organizationId,
   324→        action: "updated",
   325→        objectType: "intervention",
   326→        objectId: intervention.id,
   327→        interventionId: intervention.id,
   328→        previousState: intervention.state,
   329→        newState: "stopped",
   330→        reason: "Safety check failed at delivery time",
   331→        metadata: { checks: safetyResult.checks },
   332→      });
   333→      return { delivered: false, reason: "safety-check-failed" };
   334→    }
   335→
   336→    // ─── Generate opaque student access token ─────────────────
   337→    const tokenResult = await createStudentAccessToken({
   338→      organizationId: intervention.organizationId,
   339→      interventionId: intervention.id,
   340→      studentId: intervention.studentId,
   341→      expiresInSeconds: 7 * 24 * 60 * 60, // 7 days
   342→    });
   343→
   344→    const restPath = `rescue/${tokenResult.token}`;
   345→
   346→    // ─── Send the notification via the official API ───────────
   347→    const message = intervention.messageEdited ?? intervention.messagePreview;
   348→
   349→    try {
   350→      const sendResult = await step.run("send-notification", async () => {
   351→        // Find the course to get the externalExperienceId
   352→        const course = await db.course.findFirst({
   353→          where: { organizationId: intervention.organizationId },
   354→        });
   355→
   356→        if (!course?.externalExperienceId) {
   357→          throw new Error("Course has no externalExperienceId — cannot send notification");
   358→        }
   359→
   360→        // Call the official Whop notification API
   361→        const result = await getWhopClient().notifications.create({
   362→          experience_id: course.externalExperienceId,
   363→          title: "Continue your course",
   364→          content: message,
   365→          user_ids: [intervention.student.whopUserId],
   366→          rest_path: restPath,
   367→        });
   368→
   369→        return result;
   370→      });
   371→
   372→      // Record the delivery attempt truthfully
   373→      if (sendResult.success) {
   374→        await db.deliveryAttempt.create({
   375→          data: {
   376→            interventionId: intervention.id,
   377→            state: "api_accepted", // API accepted — NOT "delivered"
   378→            idempotencyKey,
   379→            apiAcceptedAt: new Date(),
   380→          },
   381→        });
   382→
   383→        // Set intervention state to "notification_accepted" — NOT "delivered"
   384→        // because API acceptance ≠ confirmed delivery
   385→        await db.intervention.update({
   386→          where: { id: intervention.id },
   387→          data: { state: "notification_accepted", sentAt: new Date() },
   388→        });
   389→
   390→        await recordAuditEvent({
   391→          organizationId: intervention.organizationId,
   392→          action: "delivered",
   393→          objectType: "intervention",
   394→          objectId: intervention.id,
   395→          interventionId: intervention.id,
   396→          newState: "notification_accepted",
   397→          reason: "Whop API accepted the notification",
   398→        });
   399→
   400→        return { delivered: true, apiAccepted: true };
   401→      } else {
   402→        // API returned success: false
   403→        await db.deliveryAttempt.create({
   404→          data: {
   405→            interventionId: intervention.id,
   406→            state: "failed",
   407→            idempotencyKey,
   408→            failedAt: new Date(),
   409→            errorMessage: "Whop API returned success: false",
   410→          },
   411→        });
   412→
   413→        return { delivered: false, reason: "api-rejected" };
   414→      }
   415→    } catch (error) {
   416→      // Record failure
   417→      await db.deliveryAttempt.upsert({
   418→        where: { idempotencyKey },
   419→        create: {
   420→          interventionId: intervention.id,
   421→          state: "failed",
   422→          idempotencyKey,
   423→          failedAt: new Date(),
   424→          errorMessage: error instanceof Error ? error.message : "Unknown error",
   425→        },
   426→        update: {
   427→          state: "failed",
   428→          failedAt: new Date(),
   429→          errorMessage: error instanceof Error ? error.message : "Unknown error",
   430→        },
   431→      });
   432→
   433→      throw error; // Let Inngest retry with backoff
   434→    }
   435→  },
   436→);
   437→
   438→  return [processWebhook, deliverIntervention];
   439→}

   443→function checkQuietHours(start: string, end: string, now: Date): boolean {
   444→  // Simplified: check if current hour is within the quiet period.
   445→  // Real implementation would use the student's timezone.
   446→  const currentHour = now.getHours();
   447→  const startHour = parseInt(start.split(":")[0]);
   448→  const endHour = parseInt(end.split(":")[0]);
   449→
   450→  if (startHour < endHour) {
   451→    // e.g., 12:00 - 14:00
   452→    return currentHour < startHour || currentHour >= endHour;
   453→  } else {
   454→    // Overnight: e.g., 20:00 - 08:00
   455→    return currentHour < endHour || currentHour >= startHour;
   456→  }
   457→}

   459→// ─── Typed webhook event handlers ────────────────────────────
   460→
   461→async function handleMembershipActivated(
   462→  organizationId: string,
   463→  event: WhopMembershipEvent,
   464→  webhookReceiptId: string,
   465→) {
   466→  const membership = event.data;
   467→  const whopUserId = membership.user?.id ?? membership.member?.id;
   468→
   469→  if (!whopUserId) return;
   470→
   471→  // Find or create the student
   472→  const student = await db.student.upsert({
   473→    where: {
   474→      organizationId_whopUserId: { organizationId, whopUserId },
   475→    },
   476→    create: {
   477→      organizationId,
   478→      whopUserId,
   479→    },
   480→    update: {},
   481→  });
   482→
   483→  // Find the product
   484→  const product = await db.product.findFirst({
   485→    where: { whopProductId: membership.product?.id ?? "" },
   486→  });
   487→
   488→  if (!product) return;
   489→
   490→  // Create or update the membership
   491→  await db.membership.upsert({
   492→    where: { whopMembershipId: membership.id },
   493→    create: {
   494→      organizationId,
   495→      studentId: student.id,
   496→      productId: product.id,
   497→      whopMembershipId: membership.id,
   498→      status: "active",
   499→      joinedAt: new Date(membership.created_at),
   500→      renewalDate: membership.renewal_period_end_date ? new Date(membership.renewal_period_end_date) : null,
   501→      priceCents: membership.plan?.price ?? 0,
   502→      lastSyncedAt: new Date(),
   503→    },
   504→    update: {
   505→      status: "active",
   506→      lastSyncedAt: new Date(),
   507→    },
   508→  });
   509→
   510→  // Record the event
   511→  await db.membershipEvent.create({
   512→    data: {
   513→      organizationId,
   514→      membershipId: membership.id, // Note: this is the Whop ID; the DB uses a separate ID
   515→      eventType: "activated",
   516→      occurredAt: new Date(event.timestamp),
   517→      webhookReceiptId,
   518→    },
   519→  });
   520→}

   522→async function handleMembershipDeactivated(
   523→  organizationId: string,
   524→  event: WhopMembershipEvent,
   525→  webhookReceiptId: string,
   526→) {
   527→  const membership = event.data;
   528→
   529→  // Update membership status
   530→  const dbMembership = await db.membership.findUnique({
   531→    where: { whopMembershipId: membership.id },
   532→  });
   533→
   534→  if (!dbMembership) return;
   535→
   536→  await db.membership.update({
   537→    where: { id: dbMembership.id },
   538→    data: {
   539→      status: "cancelled",
   540→      cancelledAt: new Date(event.timestamp),
   541→      lastSyncedAt: new Date(),
   542→    },
   543→  });
   544→
   545→  // Stop any pending interventions for this student
   546→  await db.intervention.updateMany({
   547→    where: {
   548→      studentId: dbMembership.studentId,
   549→      organizationId,
   550→      state: { in: ["approved", "scheduled", "queued"] },
   551→    },
   552→    data: { state: "stopped" },
   553→  });
   554→
   555→  // Record the event
   556→  await db.membershipEvent.create({
   557→    data: {
   558→      organizationId,
   559→      membershipId: dbMembership.id,
   560→      eventType: "deactivated",
   561→      occurredAt: new Date(event.timestamp),
   562→      webhookReceiptId,
   563→    },
   564→  });
   565→}

   567→async function handlePaymentSucceeded(
   568→  organizationId: string,
   569→  event: WhopPaymentEvent,
   570→  webhookReceiptId: string,
   571→) {
   572→  const payment = event.data;
   573→  const whopMembershipId = payment.membership?.id;
   574→
   575→  if (!whopMembershipId) return;
   576→
   577→  const membership = await db.membership.findUnique({
   578→    where: { whopMembershipId },
   579→  });
   580→
   581→  if (!membership) return;
   582→
   583→  // Record the payment event (idempotent via unique whopPaymentId)
   584→  await db.paymentEvent.upsert({
   585→    where: { whopPaymentId: payment.id },
   586→    create: {
   587→      organizationId,
   588→      membershipId: membership.id,
   589→      whopPaymentId: payment.id,
   590→      status: "succeeded",
   591→      amountCents: payment.amount ?? 0,
   592→      occurredAt: new Date(event.timestamp),
   593→      webhookReceiptId,
   594→    },
   595→    update: {},
   596→  });
   597→
   598→  // For Activation Rescue, ordinary payments are NOT confirmed revenue.
   599→  // Record as "observed subsequent payment" with estimated attribution.
   600→  const recentInterventions = await db.intervention.findMany({
   601→    where: {
   602→      organizationId,
   603→      studentId: membership.studentId,
   604→      state: { in: ["notification_accepted", "delivered"] },
   605→      sentAt: { lt: new Date(event.timestamp) },
   606→    },
   607→  });
   608→
   609→  for (const intervention of recentInterventions) {
   610→    // Check if course activity occurred
   611→    const progressEvent = await db.progressEvent.findFirst({
   612→      where: {
   613→        studentId: membership.studentId,
   614→        organizationId,
   615→        occurredAt: { gt: intervention.sentAt ?? new Date(0) },
   616→      },
   617→    });
   618→
   619→    await createObservedPaymentValueEvent({
   620→      organizationId,
   621→      interventionId: intervention.id,
   622→      studentId: membership.studentId,
   623→      paymentEventId: payment.id, // Unique — prevents duplicate attribution
   624→      amountCents: payment.amount ?? 0,
   625→      courseActivityOccurred: !!progressEvent,
   626→      paymentSucceededAt: new Date(event.timestamp),
   627→      interventionDeliveredAt: intervention.sentAt,
   628→    });
   629→  }
   630→}

   632→async function handleLessonCompleted(
   633→  organizationId: string,
   634→  event: WhopLessonInteractionEvent,
   635→  webhookReceiptId: string,
   636→) {
   637→  const interaction = event.data;
   638→  const whopUserId = interaction.user?.id;
   639→  const courseId = interaction.course?.id;
   640→  const lessonId = interaction.lesson?.id;
   641→  const interactionId = interaction.id; // Whop's unique interaction ID
   642→
   643→  if (!whopUserId || !courseId) return;
   644→
   645→  const student = await db.student.findFirst({
   646→    where: { organizationId, whopUserId },
   647→  });
   648→
   649→  if (!student) return;
   650→
   651→  const course = await db.course.findFirst({
   652→    where: { organizationId, externalCourseId: courseId },
   653→  });
   654→
   655→  if (!course) return;
   656→
   657→  // Record the progress event with the unique external interaction ID.
   658→  // The unique constraint prevents the same lesson interaction from being counted twice.
   659→  try {
   660→    await db.progressEvent.create({
   661→      data: {
   662→        organizationId,
   663→        studentId: student.id,
   664→        courseId: course.id,
   665→        externalInteractionId: interactionId,
   666→        lessonIndex: 0, // Whop doesn't provide an index; we use the interaction ID for uniqueness
   667→        lessonTitle: interaction.lesson?.title ?? null,
   668→        action: "completed",
   669→        occurredAt: new Date(event.timestamp),
   670→        webhookReceiptId,
   671→      },
   672→    });
   673→  } catch {
   674→    // Unique constraint violation — this interaction was already recorded.
   675→    // This is expected behavior for idempotent processing.
   676→    return;
   677→  }
   678→
   679→  // Recalculate the student's course state from authoritative data
   680→  // (don't blindly increment — recalculate from the interaction table)
   681→  const completedCount = await db.progressEvent.count({
   682→    where: {
   683→      studentId: student.id,
   684→      courseId: course.id,
   685→      action: "completed",
   686→    },
   687→  });
   688→
   689→  // Update the student course state (recalculate, don't increment)
   690→  const existingState = await db.studentCourseState.findUnique({
   691→    where: {
   692→      studentId_courseId: { studentId: student.id, courseId: course.id },
   693→    },
   694→  });
   695→
   696→  const progressPercent = course.lessonCount > 0
   697→    ? Math.round((completedCount / course.lessonCount) * 100)
   698→    : 0;
   699→
   700→  const occurredAt = new Date(event.timestamp);
   701→
   702→  // Only update lastActivityAt if the incoming event is later
   703→  // (handle out-of-order webhooks)
   704→  const shouldUpdateLastActivity =
   705→    !existingState?.lastActivityAt || occurredAt > existingState.lastActivityAt;
   706→
   707→  await db.studentCourseState.upsert({
   708→    where: {
   709→      studentId_courseId: { studentId: student.id, courseId: course.id },
   710→    },
   711→    create: {
   712→      organizationId,
   713→      studentId: student.id,
   714→      courseId: course.id,
   715→      progressPercent,
   716→      lessonsCompleted: completedCount,
   717→      totalLessons: course.lessonCount,
   718→      firstActivityAt: occurredAt,
   719→      lastActivityAt: occurredAt,
   720→    },
   721→    update: {
   722→      lessonsCompleted: completedCount, // Recalculated, not incremented
   723→      progressPercent,
   724→      lastActivityAt: shouldUpdateLastActivity ? occurredAt : undefined,
   725→    },
   726→  });
   727→
   728→  // Check for recent interventions to attribute
   729→  const recentInterventions = await db.intervention.findMany({
   730→    where: {
   731→      studentId: student.id,
   732→      organizationId,
   733→      state: { in: ["notification_accepted", "delivered"] },
   734→      sentAt: { gte: new Date(occurredAt.getTime() - 14 * 24 * 60 * 60 * 1000) },
   735→    },
   736→  });
   737→
   738→  for (const intervention of recentInterventions) {
   739→    const result = classifyActivationProgressOutcome({
   740→      interventionDeliveredAt: intervention.sentAt,
   741→      courseStartedAt: occurredAt,
   742→      progressResumedAt: occurredAt,
   743→    });
   744→
   745→    if (result.state !== "unattributed") {
   746→      // Update intervention outcome
   747→      await db.intervention.update({
   748→        where: { id: intervention.id },
   749→        data: {
   750→          outcomeState: "progress_resumed",
   751→          attributionState: result.state,
   752→          recoveredAt: occurredAt,
   753→        },
   754→      });
   755→
   756→      // Create a value event (strongly_associated, $0 — no financial value)
   757→      const valueEvent = await db.valueEvent.create({
   758→        data: {
   759→          organizationId,
   760→          interventionId: intervention.id,
   761→          studentId: student.id,
   762→          event: "Course activity resumed after notification",
   763→          attributionLevel: result.state,
   764→          amountCents: 0, // No financial value claimed
   765→          formula: result.formula,
   766→          policyVersion: result.policyVersion,
   767→        },
   768→      });
   769→
   770→      // Store the evidence chain
   771→      await db.attributionEvidence.createMany({
   772→        data: result.evidence.map((e) => ({
   773→          valueEventId: valueEvent.id,
   774→          evidenceType: e.eventType,
   775→          evidenceRef: e.timestamp,
   776→          timestamp: new Date(e.timestamp),
   777→          metadataJson: { detail: e.detail },
   778→        })),
   779→      });
   780→    }
   781→  }
   782→}

   786→/**
   787→ * Process a data export request.
   788→ * Assembles all org data and marks the request as completed.
   789→ */
   790→async function processDataExport(
   791→  exportRequestId: string,
   792→  organizationId: string,
   793→): Promise<void> {
   794→  // Mark as processing
   795→  const request = await db.dataExportRequest.findUnique({
   796→    where: { id: exportRequestId },
   797→  });
   798→
   799→  if (!request || request.status === "Completed") {
   800→    return; // Idempotent — skip if already completed or not found
   801→  }
   802→
   803→  await db.dataExportRequest.update({
   804→    where: { id: exportRequestId },
   805→    data: { status: "Processing" },
   806→  });
   807→
   808→  try {
   809→    const { assembleExport } = await import("@/lib/data-lifecycle/export-engine");
   810→    const payload = await assembleExport(organizationId);
   811→
   812→    // In production, the payload would be written to encrypted object storage
   813→    // (e.g. S3 with presigned URL). For now, we record completion and the
   814→    // download token remains valid for retrieval via a download endpoint.
   815→
   816→    await db.dataExportRequest.update({
   817→      where: { id: exportRequestId },
   818→      data: {
   819→        status: "Completed",
   820→        completedAt: new Date(),
   821→      },
   822→    });
   823→
   824→    await recordAuditEvent({
   825→      organizationId,
   826→      action: "created",
   827→      objectType: "data_export",
   828→      objectId: exportRequestId,
   829→    });
   830→  } catch (error) {
   831→    await db.dataExportRequest.update({
   832→      where: { id: exportRequestId },
   833→      data: { status: "Failed" },
   834→    });
   835→
   836→    throw error;
   837→  }
   838→}

   840→/**
   841→ * Process a data deletion request through the full lifecycle.
   842→ */
   843→async function processDataDeletion(
   844→  deletionRequestId: string,
   845→  organizationId: string,
   846→): Promise<void> {
   847→  const { verifyDeletionRequest, scheduleDeletion, executeDeletion } =
   848→    await import("@/lib/data-lifecycle/deletion-engine");
   849→
   850→  // Stage 1: Verify
   851→  const verification = await verifyDeletionRequest(deletionRequestId);
   852→  if (!verification.verified) {
   853→    return;
   854→  }
   855→
   856→  // Stage 2: Schedule (with 24h grace period)
   857→  await scheduleDeletion(deletionRequestId, 24);
   858→
   859→  // Stage 3: Execute (in production, this would be delayed by the grace period
   860→  // via Inngest's step.sleep. For now, we execute immediately after scheduling.)
   861→  const result = await executeDeletion(deletionRequestId);
   862→
   863→  await recordAuditEvent({
   864→    organizationId,
   865→    action: "deleted",
   866→    objectType: "data_deletion_request",
   867→    objectId: deletionRequestId,
   868→    newState: result.success ? "completed" : "failed",
   869→    metadata: result.evidence as Record<string, unknown>,
   870→  });
   871→}
