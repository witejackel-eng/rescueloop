   106→/**
   107→ * Classify a Prisma / DB error into a structured failure type.
   108→ * Unique constraint violations are "duplicate" — everything else is a real error.
   109→ */
   110→export function classifyCreateError(error: unknown, context: string): ClassifiedFailure {
   111→  const msg = error instanceof Error ? error.message : String(error);
   112→
   113→  // Prisma unique constraint violation codes
   114→  if (
   115→    msg.includes("Unique constraint failed") ||
   116→    msg.includes("Unique constraint violation") ||
   117→    msg.includes("P2002") // Prisma error code for unique constraint
   118→  ) {
   119→    // Determine which constraint based on context
   120→    if (context.includes("payloadHash")) {
   121→      return { class: "duplicate_payload_hash", externalId: context, message: msg };
   122→    }
   123→    return { class: "duplicate_unique_constraint", externalId: context, message: msg };
   124→  }
   125→
   126→  return { class: "db_error", externalId: context, message: msg };
   127→}

   129→// ─── Payload hash for dedup (Phase 10) ──────────────────────
   130→
   131→/**
   132→ * Simple deterministic hash for progress event dedup.
   133→ * Produces a hex string from the interaction's identifying fields.
   134→ * In production, replace with crypto.subtle.digest("SHA-256", ...).
   135→ */
   136→export function computePayloadHash(fields: {
   137→  userId: string;
   138→  courseId: string;
   139→  lessonId: string;
   140→  action: string;
   141→  sourceTimestamp: string;
   142→}): string {
   143→  // Deterministic concatenation for hashing
   144→  const raw = `${fields.userId}|${fields.courseId}|${fields.lessonId}|${fields.action}|${fields.sourceTimestamp}`;
   145→  // Simple FNV-1a hash for non-crypto dedup (good enough for idempotency keys)
   146→  let hash = 0x811c9dc5;
   147→  for (let i = 0; i < raw.length; i++) {
   148→    hash ^= raw.charCodeAt(i);
   149→    hash = Math.imul(hash, 0x01000193);
   150→  }
   151→  // Convert to unsigned 32-bit hex
   152→  return (hash >>> 0).toString(16).padStart(8, "0");
   153→}

   155→// ─── Bounded concurrency helper ──────────────────────────────
   156→
   157→/** Run items with at most `concurrency` operations in flight. */
   158→async function boundedMap<T, R>(
   159→  items: T[],
   160→  fn: (item: T) => Promise<R>,
   161→  concurrency: number = MAX_CONCURRENT_UPDATES,
   162→): Promise<R[]> {
   163→  const results: R[] = [];
   164→  const executing: Promise<void>[] = [];
   165→
   166→  for (const item of items) {
   167→    const p = fn(item).then((r) => { results.push(r); });
   168→    executing.push(p);
   169→
   170→    if (executing.length >= concurrency) {
   171→      await Promise.race(executing);
   172→      // Remove settled promises
   173→      for (let i = executing.length - 1; i >= 0; i--) {
   174→        const settled = await Promise.race([
   175→          executing[i].then(() => true, () => true),
   176→          Promise.resolve(false),
   177→        ]);
   178→        if (settled) executing.splice(i, 1);
   179→      }
   180→    }
   181→  }
   182→
   183→  await Promise.allSettled(executing);
   184→  return results;
   185→}

   189→export async function syncCourses(params: SyncExecutionParams): Promise<SyncResult> {
   190→  const result: SyncResult = {
   191→    resource: "courses",
   192→    recordsRead: 0,
   193→    recordsCreated: 0,
   194→    recordsUpdated: 0,
   195→    recordsSkipped: 0,
   196→    errors: [],
   197→    warnings: [],
   198→    cursor: null,
   199→  };
   200→
   201→  // Resume from checkpoint if no explicit cursor
   202→  let cursor = params.cursor ?? null;
   203→  if (!cursor) {
   204→    const checkpoint = await getLatestCheckpoint(params.organizationId, "courses");
   205→    if (checkpoint?.cursor) {
   206→      cursor = checkpoint.cursor;
   207→    }
   208→  }
   209→
   210→  let pagesProcessed = 0;
   211→
   212→  do {
   213→    const page = await params.providers.courses.list({
   214→      companyId: params.companyId,
   215→      cursor,
   216→      pageSize: PAGE_SIZE,
   217→    });
   218→
   219→    result.recordsRead += page.items.length;
   220→    pagesProcessed++;
   221→
   222→    // 1. Validate the full page
   223→    const validItems = page.items.filter((course) => {
   224→      if (!course.id) {
   225→        result.errors.push(`Course missing id — skipped`);
   226→        result.recordsSkipped++;
   227→        return false;
   228→      }
   229→      return true;
   230→    });
   231→
   232→    if (validItems.length === 0) {
   233→      cursor = page.nextCursor;
   234→      continue;
   235→    }
   236→
   237→    // 2. Load existing external IDs in one bounded query
   238→    const externalIds = validItems.map((c) => c.id);
   239→    const existing = await db.course.findMany({
   240→      where: {
   241→        organizationId: params.organizationId,
   242→        externalCourseId: { in: externalIds },
   243→      },
   244→      select: { id: true, externalCourseId: true },
   245→    });
   246→    const existingMap = new Map(existing.map((e) => [e.externalCourseId, e.id]));
   247→
   248→    // 3. Split into creates and updates
   249→    const toCreate = validItems.filter((c) => !existingMap.has(c.id));
   250→    const toUpdate = validItems.filter((c) => existingMap.has(c.id));
   251→
   252→    // 4. Batch creates with skipDuplicates
   253→    if (toCreate.length > 0) {
   254→      try {
   255→        await db.course.createMany({
   256→          data: toCreate.map((course) => ({
   257→            organizationId: params.organizationId,
   258→            externalCourseId: course.id,
   259→            externalExperienceId: course.experienceId,
   260→            name: course.title ?? "Untitled Course",
   261→            lessonCount: course.lessonCount,
   262→          })),
   263→          skipDuplicates: true,
   264→        });
   265→        result.recordsCreated += toCreate.length;
   266→      } catch (error) {
   267→        result.errors.push(`Course batch create: ${error instanceof Error ? error.message : "unknown"}`);
   268→        result.recordsSkipped += toCreate.length;
   269→      }
   270→    }
   271→
   272→    // 5. Batch updates with bounded concurrency
   273→    if (toUpdate.length > 0) {
   274→      const updateResults = await boundedMap(toUpdate, async (course) => {
   275→        const existingId = existingMap.get(course.id)!;
   276→        try {
   277→          await db.course.update({
   278→            where: { id: existingId },
   279→            data: {
   280→              name: course.title ?? undefined,
   281→              lessonCount: course.lessonCount || undefined,
   282→              externalExperienceId: course.experienceId || undefined,
   283→            },
   284→          });
   285→          return true;
   286→        } catch (error) {
   287→          result.errors.push(`Course ${course.id}: ${error instanceof Error ? error.message : "unknown"}`);
   288→          return false;
   289→        }
   290→      });
   291→      result.recordsUpdated += updateResults.filter(Boolean).length;
   292→      result.recordsSkipped += updateResults.filter((r) => !r).length;
   293→    }
   294→
   295→    // 6. Commit page — persist checkpoint
   296→    cursor = page.nextCursor;
   297→    if (params.syncExecutionId) {
   298→      await persistCheckpoint(
   299→        params.organizationId,
   300→        "courses",
   301→        cursor,
   302→        null,
   303→        params.syncExecutionId,
   304→        pagesProcessed,
   305→      );
   306→    }
   307→  } while (cursor);
   308→
   309→  result.cursor = cursor;
   310→
   311→  // 7. Emit usage and audit data
   312→  await recordAuditEvent({
   313→    organizationId: params.organizationId,
   314→    action: "synced",
   315→    objectType: "course",
   316→    objectId: "bulk",
   317→    reason: `Synced ${result.recordsRead} courses`,
   318→    metadata: {
   319→      created: result.recordsCreated,
   320→      updated: result.recordsUpdated,
   321→      skipped: result.recordsSkipped,
   322→      errors: result.errors.length,
   323→    },
   324→  });
   325→
   326→  return result;
   327→}

   470→export async function syncMemberships(params: SyncExecutionParams): Promise<SyncResult> {
   471→  const result: SyncResult = {
   472→    resource: "memberships",
   473→    recordsRead: 0,
   474→    recordsCreated: 0,
   475→    recordsUpdated: 0,
   476→    recordsSkipped: 0,
   477→    errors: [],
   478→    warnings: [],
   479→    cursor: null,
   480→  };
   481→
   482→  let cursor = params.cursor ?? null;
   483→  if (!cursor) {
   484→    const checkpoint = await getLatestCheckpoint(params.organizationId, "memberships");
   485→    if (checkpoint?.cursor) {
   486→      cursor = checkpoint.cursor;
   487→    }
   488→  }
   489→
   490→  let pagesProcessed = 0;
   491→
   492→  do {
   493→    const page = await params.providers.memberships.list({
   494→      companyId: params.companyId,
   495→      cursor,
   496→      pageSize: PAGE_SIZE,
   497→    });
   498→
   499→    result.recordsRead += page.items.length;
   500→    pagesProcessed++;
   501→
   502→    for (const membership of page.items) {
   503→      try {
   504→        // Normalize the external status — replaces `membership.status as any`
   505→        const { status: safeStatus, warning } = normalizeMembershipStatus(membership.status);
   506→        if (warning) {
   507→          result.warnings.push(warning);
   508→        }
   509→
   510→        // Find or create the student (upsert by composite key)
   511→        const student = await db.student.upsert({
   512→          where: {
   513→            organizationId_whopUserId: {
   514→              organizationId: params.organizationId,
   515→              whopUserId: membership.userId,
   516→            },
   517→          },
   518→          create: {
   519→            organizationId: params.organizationId,
   520→            whopUserId: membership.userId,
   521→          },
   522→          update: {},
   523→        });
   524→
   525→        // Find the product
   526→        const product = await db.product.findFirst({
   527→          where: {
   528→            organizationId: params.organizationId,
   529→            whopProductId: membership.productId,
   530→          },
   531→        });
   532→
   533→        if (!product) {
   534→          result.errors.push(`Membership ${membership.id}: product ${membership.productId} not found`);
   535→          result.recordsSkipped++;
   536→          continue;
   537→        }
   538→
   539→        // Check if membership already exists to determine create vs update
   540→        const existingMembership = await db.membership.findUnique({
   541→          where: { whopMembershipId: membership.id },
   542→          select: { id: true },
   543→        });
   544→
   545→        if (existingMembership) {
   546→          // UPDATE — increment recordsUpdated, NOT recordsCreated
   547→          await db.membership.update({
   548→            where: { id: existingMembership.id },
   549→            data: {
   550→              status: safeStatus,
   551→              renewalDate: membership.renewalDate ? new Date(membership.renewalDate) : null,
   552→              cancelledAt: membership.cancelledAt ? new Date(membership.cancelledAt) : null,
   553→              priceCents: membership.priceCents,
   554→              currency: membership.currency,
   555→              lastSyncedAt: new Date(),
   556→            },
   557→          });
   558→          result.recordsUpdated++;
   559→        } else {
   560→          // CREATE
   561→          await db.membership.create({
   562→            data: {
   563→              organizationId: params.organizationId,
   564→              studentId: student.id,
   565→              productId: product.id,
   566→              whopMembershipId: membership.id,
   567→              status: safeStatus,
   568→              joinedAt: new Date(membership.joinedAt),
   569→              renewalDate: membership.renewalDate ? new Date(membership.renewalDate) : null,
   570→              cancelledAt: membership.cancelledAt ? new Date(membership.cancelledAt) : null,
   571→              priceCents: membership.priceCents,
   572→              currency: membership.currency,
   573→              lastSyncedAt: new Date(),
   574→            },
   575→          });
   576→          result.recordsCreated++;
   577→        }
   578→      } catch (error) {
   579→        const failure = classifyCreateError(error, `membership-${membership.id}`);
   580→        if (failure.class === "duplicate_unique_constraint") {
   581→          // Idempotent — already ingested
   582→          result.recordsSkipped++;
   583→        } else {
   584→          result.errors.push(`Membership ${membership.id}: ${failure.message}`);
   585→          result.recordsSkipped++;
   586→        }
   587→      }
   588→    }
   589→
   590→    // Persist checkpoint
   591→    cursor = page.nextCursor;
   592→    if (params.syncExecutionId) {
   593→      await persistCheckpoint(
   594→        params.organizationId,
   595→        "memberships",
   596→        cursor,
   597→        null,
   598→        params.syncExecutionId,
   599→        pagesProcessed,
   600→      );
   601→    }
   602→  } while (cursor);
   603→
   604→  result.cursor = cursor;
   605→
   606→  await recordAuditEvent({
   607→    organizationId: params.organizationId,
   608→    action: "synced",
   609→    objectType: "membership",
   610→    objectId: "bulk",
   611→    reason: `Synced ${result.recordsRead} memberships`,
   612→    metadata: {
   613→      created: result.recordsCreated,
   614→      updated: result.recordsUpdated,
   615→      skipped: result.recordsSkipped,
   616→      warnings: result.warnings.length,
   617→    },
   618→  });
   619→
   620→  return result;
   621→}

   623→// ─── Course progress sync (Phase 10: idempotent + bounded) ──

   625→export async function syncCourseProgress(
   626→  params: SyncExecutionParams & { courseId: string },
   627→): Promise<SyncResult> {
   628→  const result: SyncResult = {
   629→    resource: "progress",
   630→    recordsRead: 0,
   631→    recordsCreated: 0,
   632→    recordsUpdated: 0,
   633→    recordsSkipped: 0,
   634→    errors: [],
   635→    warnings: [],
   636→    cursor: null,
   637→  };
   638→
   639→  // Find the internal course
   640→  const course = await db.course.findFirst({
   641→    where: {
   642→      organizationId: params.organizationId,
   643→      id: params.courseId,
   644→    },
   645→  });
   646→
   647→  if (!course) {
   648→    result.errors.push(`Course ${params.courseId} not found`);
   649→    return result;
   650→  }
   651→
   652→  let cursor = params.cursor ?? null;
   653→  if (!cursor) {
   654→    const checkpoint = await getLatestCheckpoint(params.organizationId, "progress");
   655→    if (checkpoint?.cursor) {
   656→      cursor = checkpoint.cursor;
   657→    }
   658→  }
   659→
   660→  let pagesProcessed = 0;
   661→  const now = new Date();
   662→
   663→  do {
   664→    const page = await params.providers.progress.listLessonInteractions({
   665→      companyId: params.companyId,
   666→      courseId: course.externalCourseId,
   667→      cursor,
   668→      pageSize: PAGE_SIZE,
   669→    });
   670→
   671→    result.recordsRead += page.items.length;
   672→    pagesProcessed++;
   673→
   674→    // Phase 10: Resolve all students for this page in one query
   675→    const userIds = [...new Set(page.items.map((i) => i.userId))];
   676→    const students = await db.student.findMany({
   677→      where: {
   678→        organizationId: params.organizationId,
   679→        whopUserId: { in: userIds },
   680→      },
   681→      select: { id: true, whopUserId: true },
   682→    });
   683→    const studentMap = new Map(students.map((s) => [s.whopUserId, s.id]));
   684→
   685→    // Phase 10: Check existing interaction IDs to split creates from skips
   686→    const interactionIds = page.items
   687→      .map((i) => i.id)
   688→      .filter(Boolean);
   689→    const existingEvents = interactionIds.length > 0
   690→      ? await db.progressEvent.findMany({
   691→          where: {
   692→            organizationId: params.organizationId,
   693→            externalInteractionId: { in: interactionIds },
   694→          },
   695→          select: { externalInteractionId: true },
   696→        })
   697→      : [];
   698→    const existingInteractionSet = new Set(existingEvents.map((e) => e.externalInteractionId));
   699→
   700→    // Phase 10: Also check existing payload hashes for dedup
   701→    const payloadHashes = page.items.map((interaction) =>
   702→      computePayloadHash({
   703→        userId: interaction.userId,
   704→        courseId: interaction.courseId,
   705→        lessonId: interaction.lessonId,
   706→        action: interaction.completed ? "completed" : "started",
   707→        sourceTimestamp: interaction.sourceTimestamp,
   708→      }),
   709→    );
   710→    const existingByHash = payloadHashes.length > 0
   711→      ? await db.progressEvent.findMany({
   712→          where: {
   713→            organizationId: params.organizationId,
   714→            payloadHash: { in: payloadHashes },
   715→          },
   716→          select: { payloadHash: true },
   717→        })
   718→      : [];
   719→    const existingHashSet = new Set(existingByHash.map((e) => e.payloadHash));
   720→
   721→    // Build create data, filtering out duplicates and missing students
   722→    const toCreate: Array<{
   723→      organizationId: string;
   724→      studentId: string;
   725→      courseId: string;
   726→      externalInteractionId: string;
   727→      lessonId: string;
   728→      lessonIndex: number;
   729→      lessonTitle: string | null;
   730→      action: string;
   731→      payloadHash: string;
   732→      sourceTimestamp: Date;
   733→      receivedAt: Date;
   734→      sourceVersion: string | null;
   735→      occurredAt: Date;
   736→    }> = [];
   737→
   738→    for (const interaction of page.items) {
   739→      // Skip if student not found — students with activity but no qualifying membership
   740→      const studentId = studentMap.get(interaction.userId);
   741→      if (!studentId) {
   742→        result.warnings.push(
   743→          `Interaction ${interaction.id}: student ${interaction.userId} not found — skipped (membership may arrive later)`,
   744→        );
   745→        result.recordsSkipped++;
   746→        continue;
   747→      }
   748→
   749→      // Phase 10: Skip duplicates by external interaction ID
   750→      if (existingInteractionSet.has(interaction.id)) {
   751→        result.recordsSkipped++;
   752→        continue;
   753→      }
   754→
   755→      // Phase 10: Compute payload hash and skip if already ingested
   756→      const hash = computePayloadHash({
   757→        userId: interaction.userId,
   758→        courseId: interaction.courseId,
   759→        lessonId: interaction.lessonId,
   760→        action: interaction.completed ? "completed" : "started",
   761→        sourceTimestamp: interaction.sourceTimestamp,
   762→      });
   763→
   764→      if (existingHashSet.has(hash)) {
   765→        result.recordsSkipped++;
   766→        continue;
   767→      }
   768→
   769→      toCreate.push({
   770→        organizationId: params.organizationId,
   771→        studentId,
   772→        courseId: course.id,
   773→        externalInteractionId: interaction.id,
   774→        lessonId: interaction.lessonId,
   775→        lessonIndex: 0, // Lesson index not available from provider; use 0 as placeholder
   776→        lessonTitle: interaction.lessonTitle,
   777→        action: interaction.completed ? "completed" : "started",
   778→        payloadHash: hash,
   779→        sourceTimestamp: new Date(interaction.sourceTimestamp),
   780→        receivedAt: now,
   781→        sourceVersion: null, // No version from current provider contract
   782→        occurredAt: new Date(interaction.createdAt),
   783→      });
   784→    }
   785→
   786→    // Phase 10: Batch create with skipDuplicates for true idempotency
   787→    if (toCreate.length > 0) {
   788→      // Process in sub-batches to avoid oversized queries
   789→      for (let i = 0; i < toCreate.length; i += PROGRESS_BATCH_SIZE) {
   790→        const batch = toCreate.slice(i, i + PROGRESS_BATCH_SIZE);
   791→        try {
   792→          const createResult = await db.progressEvent.createMany({
   793→            data: batch,
   794→            skipDuplicates: true,
   795→          });
   796→          result.recordsCreated += createResult.count;
   797→          // Any items not created were duplicates (skipDuplicates silently skips)
   798→          result.recordsSkipped += batch.length - createResult.count;
   799→        } catch (error) {
   800→          const failure = classifyCreateError(error, "progress-batch");
   801→          if (
   802→            failure.class === "duplicate_unique_constraint" ||
   803→            failure.class === "duplicate_payload_hash"
   804→          ) {
   805→            // These are expected under race conditions — all treated as skipped
   806→            result.recordsSkipped += batch.length;
   807→          } else {
   808→            result.errors.push(`Progress batch create: ${failure.message}`);
   809→            result.recordsSkipped += batch.length;
   810→          }
   811→        }
   812→      }
   813→    }
   814→
   815→    // Persist checkpoint
   816→    cursor = page.nextCursor;
   817→    if (params.syncExecutionId) {
   818→      await persistCheckpoint(
   819→        params.organizationId,
   820→        "progress",
   821→        cursor,
   822→        null,
   823→        params.syncExecutionId,
   824→        pagesProcessed,
   825→      );
   826→    }
   827→  } while (cursor);
   828→
   829→  // Phase 10: Recalculate course states using bounded aggregation
   830→  await recalculateCourseStatesBounded(params.organizationId, course.id);
   831→
   832→  result.cursor = cursor;
   833→
   834→  await recordAuditEvent({
   835→    organizationId: params.organizationId,
   836→    action: "synced",
   837→    objectType: "progress",
   838→    objectId: course.id,
   839→    reason: `Synced ${result.recordsRead} progress events`,
   840→    metadata: {
   841→      created: result.recordsCreated,
   842→      skipped: result.recordsSkipped,
   843→      duplicatesHandled: result.recordsSkipped,
   844→    },
   845→  });
   846→
   847→  return result;
   848→}
   849→
   850→// ─── Reconciliation (Phase 11: set-based queries) ───────────
   851→
   852→export interface ReconciliationResult {
   853→  matched: number;
   854→  membershipWithoutCourseActivity: number;
   855→  courseActivityWithoutMembership: number;
   856→  unmappedProduct: number;
   857→  missingSourceFields: number;
   858→  staleSourceRecord: number;
   859→  totalEvaluated: number;
   860→}
   861→
   862→export async function reconcile(
   863→  organizationId: string,
   864→  courseId: string,
   865→  syncExecutionId?: string,
   866→): Promise<ReconciliationResult> {
   867→  const result: ReconciliationResult = {
   868→    matched: 0,
   869→    membershipWithoutCourseActivity: 0,
   870→    courseActivityWithoutMembership: 0,
   871→    unmappedProduct: 0,
   872→    missingSourceFields: 0,
   873→    staleSourceRecord: 0,
   874→    totalEvaluated: 0,
   875→  };
   876→
   877→  // Create a reconciliation run record
   878→  const run = await createReconciliationRun(organizationId, courseId, syncExecutionId);
   879→
   880→  // Phase 11: Set-based queries instead of per-student N+1
   881→
   882→  // 1. Get all confirmed mappings for this course (bounded — typically ≤10 products per course)
   883→  const mappings = await db.productCourseMapping.findMany({
   884→    where: { organizationId, courseId, isConfirmed: true },
   885→    select: { id: true, productId: true, courseId: true },
   886→  });
   887→
   888→  const mappedProductIds = new Set(mappings.map((m) => m.productId));
   889→
   890→  // 2. Get all student IDs with qualifying memberships for mapped products (ONE query)
   891→  const studentsWithMembership = await db.membership.findMany({
   892→    where: {
   893→      organizationId,
   894→      productId: { in: Array.from(mappedProductIds) },
   895→    },
   896→    select: { studentId: true, id: true, productId: true },
   897→  });
   898→
   899→  // Build map: studentId → membership info for classification
   900→  const membershipByStudent = new Map<string, { membershipId: string; productId: string }[]>();
   901→  for (const m of studentsWithMembership) {
   902→    const existing = membershipByStudent.get(m.studentId) ?? [];
   903→    existing.push({ membershipId: m.id, productId: m.productId });
   904→    membershipByStudent.set(m.studentId, existing);
   905→  }
   906→
   907→  const studentIdsWithMembership = new Set(membershipByStudent.keys());
   908→
   909→  // 3. Get all student IDs with course activity (ONE query)
   910→  const courseStates = await db.studentCourseState.findMany({
   911→    where: { organizationId, courseId },
   912→    select: { studentId: true, id: true, lessonsCompleted: true },
   913→  });
   914→
   915→  const studentIdsWithActivity = new Set(courseStates.map((s) => s.studentId));
   916→
   917→  // 4. Set-based classification — no per-student queries
   918→  // Matched: students in both sets
   919→  const matchedStudentIds = new Set(
   920→    [...studentIdsWithMembership].filter((id) => studentIdsWithActivity.has(id)),
   921→  );
   922→
   923→  // Membership without course activity
   924→  const membershipOnlyStudentIds = new Set(
   925→    [...studentIdsWithMembership].filter((id) => !studentIdsWithActivity.has(id)),
   926→  );
   927→
   928→  // Course activity without membership
   929→  const activityOnlyStudentIds = new Set(
   930→    [...studentIdsWithActivity].filter((id) => !studentIdsWithMembership.has(id)),
   931→  );
   932→
   933→  result.matched = matchedStudentIds.size;
   934→  result.membershipWithoutCourseActivity = membershipOnlyStudentIds.size;
   935→  result.courseActivityWithoutMembership = activityOnlyStudentIds.size;
   936→  result.totalEvaluated = studentIdsWithMembership.size + activityOnlyStudentIds.size;
   937→
   938→  // 5. Persist reconciliation outcomes in batches
   939→  const outcomeBatches: Array<{
   940→    reconciliationRunId: string;
   941→    organizationId: string;
   942→    studentId: string;
   943→    membershipId: string | null;
   944→    courseId: string;
   945→    mappingId: string | null;
   946→    classification: string;
   947→    evidenceJson: unknown;
   948→  }> = [];
   949→
   950→  // Matched outcomes
   951→  for (const studentId of matchedStudentIds) {
   952→    const memberships = membershipByStudent.get(studentId) ?? [];
   953→    const mappingId = mappings.find((m) =>
   954→      memberships.some((mem) => mem.productId === m.productId),
   955→    )?.id ?? null;
   956→    outcomeBatches.push({
   957→      reconciliationRunId: run.id,
   958→      organizationId,
   959→      studentId,
   960→      membershipId: memberships[0]?.membershipId ?? null,
   961→      courseId,
   962→      mappingId,
   963→      classification: "matched",
   964→      evidenceJson: { hasMembership: true, hasCourseActivity: true },
   965→    });
   966→  }
   967→
   968→  // Membership without course activity
   969→  for (const studentId of membershipOnlyStudentIds) {
   970→    const memberships = membershipByStudent.get(studentId) ?? [];
   971→    const mappingId = mappings.find((m) =>
   972→      memberships.some((mem) => mem.productId === m.productId),
   973→    )?.id ?? null;
   974→    outcomeBatches.push({
   975→      reconciliationRunId: run.id,
   976→      organizationId,
   977→      studentId,
   978→      membershipId: memberships[0]?.membershipId ?? null,
   979→      courseId,
   980→      mappingId,
   981→      classification: "membership_without_course_activity",
   982→      evidenceJson: { hasMembership: true, hasCourseActivity: false },
   983→    });
   984→  }
   985→
   986→  // Course activity without membership
   987→  for (const studentId of activityOnlyStudentIds) {
   988→    outcomeBatches.push({
   989→      reconciliationRunId: run.id,
   990→      organizationId,
   991→      studentId,
   992→      membershipId: null,
   993→      courseId,
   994→      mappingId: null,
   995→      classification: "course_activity_without_membership",
   996→      evidenceJson: { hasMembership: false, hasCourseActivity: true },
   997→    });
   998→  }
   999→
  1000→  // Batch persist outcomes
  1001→  for (let i = 0; i < outcomeBatches.length; i += RECONCILIATION_PAGE_SIZE) {
  1002→    const batch = outcomeBatches.slice(i, i + RECONCILIATION_PAGE_SIZE);
  1003→    await db.reconciliationOutcome.createMany({
  1004→      data: batch.map((o) => ({
  1005→        reconciliationRunId: o.reconciliationRunId,
  1006→        organizationId: o.organizationId,
  1007→        studentId: o.studentId,
  1008→        membershipId: o.membershipId,
  1009→        courseId: o.courseId,
  1010→        mappingId: o.mappingId,
  1011→        classification: o.classification as any,
  1012→        evidenceJson: o.evidenceJson as any,
  1013→      })),
  1014→    });
  1015→  }
  1016→
  1017→  // Complete the reconciliation run record
  1018→  await completeReconciliationRun(run.id, {
  1019→    matched: result.matched,
  1020→    membershipWithoutCourseActivity: result.membershipWithoutCourseActivity,
  1021→    courseActivityWithoutMembership: result.courseActivityWithoutMembership,
  1022→    unmappedProduct: result.unmappedProduct,
  1023→    missingSourceFields: result.missingSourceFields,
  1024→    staleSourceRecord: result.staleSourceRecord,
  1025→  });
  1026→
  1027→  return result;
  1028→}

  1030→// ─── Candidate detection (Phase 12: full eligibility checks) ─
  1031→
  1032→export interface CandidateDetectionResult {
  1033→  candidatesFound: number;
  1034→  snapshotsCreated: number;
  1035→  snapshotsSkipped: number;
  1036→  errors: string[];
  1037→  checksPerformed: number;
  1038→}
  1039→
  1040→/**
  1041→ * Detect candidates for an Activation Rescue campaign.
  1042→ *
  1043→ * Phase 12: Operates ONLY on the campaign's confirmed mapping.
  1044→ * All 17 eligibility checks are verified before creating a snapshot.
  1045→ * Uses batch queries — no N+1 patterns.
  1046→ */
  1047→export async function detectCandidates(
  1048→  organizationId: string,
  1049→  campaignId: string,
  1050→): Promise<CandidateDetectionResult> {
  1051→  const result: CandidateDetectionResult = {
  1052→    candidatesFound: 0,
  1053→    snapshotsCreated: 0,
  1054→    snapshotsSkipped: 0,
  1055→    errors: [],
  1056→    checksPerformed: 0,
  1057→  };
  1058→
  1059→  const now = new Date();
  1060→
  1061→  // ─── Pre-flight checks (fail fast) ────────────────────────
  1062→
  1063→  // Fetch campaign with organization and latest version
  1064→  const campaign = await db.campaign.findFirst({
  1065→    where: { id: campaignId, organizationId },
  1066→    include: {
  1067→      organization: {
  1068→        include: { installations: true },
  1069→      },
  1070→      versions: { orderBy: { versionNumber: "desc" }, take: 1 },
  1071→      confirmedMapping: {
  1072→        include: {
  1073→          product: true,
  1074→          course: true,
  1075→        },
  1076→      },
  1077→    },
  1078→  });
  1079→
  1080→  // Check: Campaign exists and is Activation Rescue
  1081→  if (!campaign || campaign.type !== "activation_rescue") {
  1082→    result.errors.push("Campaign not found or not an Activation Rescue campaign");
  1083→    return result;
  1084→  }
  1085→
  1086→  const latestVersion = campaign.versions[0];
  1087→  const org = campaign.organization;
  1088→
  1089→  // Check 1: Organisation active
  1090→  if (org.status !== "active") {
  1091→    result.errors.push(`Organisation status is "${org.status}", not "active"`);
  1092→    return result;
  1093→  }
  1094→
  1095→  // Check 2: Organisation not paused
  1096→  if (org.isPaused) {
  1097→    result.errors.push("Organisation is paused");
  1098→    return result;
  1099→  }
  1100→
  1101→  // Check 3: Installation active
  1102→  const activeInstallation = org.installations.find((i) => i.status === "active");
  1103→  if (!activeInstallation) {
  1104→    result.errors.push("No active installation");
  1105→    return result;
  1106→  }
  1107→
  1108→  // Check 4: Campaign active
  1109→  if (campaign.status !== "active") {
  1110→    result.errors.push(`Campaign status is "${campaign.status}", not "active"`);
  1111→    return result;
  1112→  }
  1113→
  1114→  // Check 5: Campaign is Activation Rescue (already verified above)
  1115→  // Check 6: Manual approval enabled
  1116→  if (campaign.approvalMode !== "manual") {
  1117→    result.errors.push("Campaign does not have manual approval enabled");
  1118→    return result;
  1119→  }
  1120→
  1121→  // Check 7: Campaign version exists
  1122→  if (!latestVersion) {
  1123→    result.errors.push("No campaign version exists");
  1124→    return result;
  1125→  }
  1126→
  1127→  // Check 8: Confirmed mapping belongs to campaign
  1128→  const mapping = campaign.confirmedMapping;
  1129→  if (!mapping) {
  1130→    result.errors.push("Campaign has no confirmed mapping");
  1131→    return result;
  1132→  }
  1133→
  1134→  if (!mapping.isConfirmed) {
  1135→    result.errors.push("Campaign's mapping is not confirmed");
  1136→    return result;
  1137→  }
  1138→
  1139→  const courseId = mapping.courseId;
  1140→  const productId = mapping.productId;
  1141→
  1142→  // ─── Batch queries (no N+1) ───────────────────────────────
  1143→
  1144→  // All memberships for the mapped product that are active or trialing
  1145→  const qualifyingMemberships = await db.membership.findMany({
  1146→    where: {
  1147→      organizationId,
  1148→      productId,
  1149→      status: { in: ["active", "trialing"] },
  1150→    },
  1151→    include: { student: true },
  1152→  });
  1153→
  1154→  if (qualifyingMemberships.length === 0) {
  1155→    return result; // No candidates possible
  1156→  }
  1157→
  1158→  const candidateStudentIds = new Set(qualifyingMemberships.map((m) => m.studentId));
  1159→
  1160→  // Batch: course activity states for all candidate students
  1161→  const courseStates = await db.studentCourseState.findMany({
  1162→    where: {
  1163→      organizationId,
  1164→      courseId,
  1165→      studentId: { in: Array.from(candidateStudentIds) },
  1166→    },
  1167→  });
  1168→  const activityByStudent = new Map(courseStates.map((s) => [s.studentId, s]));
  1169→
  1170→  // Batch: suppressions for all candidate students (course, campaign, and org level)
  1171→  const suppressions = await db.suppression.findMany({
  1172→    where: {
  1173→      organizationId,
  1174→      studentId: { in: Array.from(candidateStudentIds) },
  1175→    },
  1176→  });
  1177→  const suppressedStudentIds = new Set(suppressions.map((s) => s.studentId));
  1178→
  1179→  // Batch: existing active interventions for these students on this campaign
  1180→  const activeInterventions = await db.intervention.findMany({
  1181→    where: {
  1182→      organizationId,
  1183→      studentId: { in: Array.from(candidateStudentIds) },
  1184→      campaignId,
  1185→      state: { notIn: ["dismissed", "stopped", "failed"] },
  1186→    },
  1187→  });
  1188→  const studentsWithActiveIntervention = new Set(activeInterventions.map((i) => i.studentId));
  1189→
  1190→  // Batch: last intervention per student for cooldown check
  1191→  const lastInterventions = await db.intervention.findMany({
  1192→    where: {
  1193→      organizationId,
  1194→      studentId: { in: Array.from(candidateStudentIds) },
  1195→      campaignId,
  1196→    },
  1197→    orderBy: { createdAt: "desc" },
  1198→    // We'll process these to get the latest per student
  1199→  });
  1200→  const lastInterventionByStudent = new Map<string, Date>();
  1201→  for (const iv of lastInterventions) {
  1202→    if (!lastInterventionByStudent.has(iv.studentId)) {
  1203→      lastInterventionByStudent.set(iv.studentId, iv.createdAt);
  1204→    }
  1205→  }
  1206→
  1207→  // Batch: recent intervention counts per student (for per-student message limit)
  1208→  const thirtyDaysAgo = new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000);
  1209→  const recentInterventions = await db.intervention.findMany({
  1210→    where: {
  1211→      organizationId,
  1212→      studentId: { in: Array.from(candidateStudentIds) },
  1213→      campaignId,
  1214→      createdAt: { gte: thirtyDaysAgo },
  1215→      state: { in: ["notification_accepted", "delivered"] },
  1216→    },
  1217→    select: { studentId: true },
  1218→  });
  1219→  const recentInterventionCounts = new Map<string, number>();
  1220→  for (const iv of recentInterventions) {
  1221→    recentInterventionCounts.set(iv.studentId, (recentInterventionCounts.get(iv.studentId) ?? 0) + 1);
  1222→  }
  1223→
  1224→  // Batch: org-wide message count in last 30 days
  1225→  const orgWideMessageCount = await db.intervention.count({
  1226→    where: {
  1227→      organizationId,
  1228→      createdAt: { gte: thirtyDaysAgo },
  1229→      state: { in: ["notification_accepted", "delivered"] },
  1230→    },
  1231→  });
  1232→
  1233→  // Batch: campaign message count in last 30 days
  1234→  const campaignMessageCount = await db.intervention.count({
  1235→    where: {
  1236→      organizationId,
  1237→      campaignId,
  1238→      createdAt: { gte: thirtyDaysAgo },
  1239→      state: { in: ["notification_accepted", "delivered"] },
  1240→    },
  1241→  });
  1242→
  1243→  // Batch: existing eligibility snapshots for this campaign version + window
  1244→  // (to prevent duplicate snapshots on re-run)
  1245→  const eligibilityWindowStart = new Date(
  1246→    now.getFullYear(),
  1247→    now.getMonth(),
  1248→    now.getDate(),
  1249→  ); // Start of today as the eligibility window
  1250→
  1251→  const existingSnapshots = await db.eligibilitySnapshot.findMany({
  1252→    where: {
  1253→      organizationId,
  1254→      studentId: { in: Array.from(candidateStudentIds) },
  1255→      campaignVersionId: latestVersion.id,
  1256→      eligibilityWindowStart,
  1257→    },
  1258→    select: { studentId: true },
  1259→  });
  1260→  const studentsWithExistingSnapshot = new Set(existingSnapshots.map((s) => s.studentId));
  1261→
  1262→  // Batch: check plan allows monitored members
  1263→  const entitlement = await db.subscriptionEntitlement.findFirst({
  1264→    where: { organizationId },
  1265→  });
  1266→  const plan = entitlement
  1267→    ? await db.plan.findUnique({ where: { tier: entitlement.planTier } })
  1268→    : null;
  1269→  const maxMonitoredMembers = plan?.maxMonitoredMembers ?? Infinity;
  1270→
  1271→  // Count current monitored members
  1272→  const currentMonitoredMembers = await db.membership.count({
  1273→    where: {
  1274→      organizationId,
  1275→      status: { in: ["active", "trialing"] },
  1276→    },
  1277→  });
  1278→
  1279→  // Batch: source data freshness check
  1280→  const latestCheckpoint = await db.syncCheckpoint.findFirst({
  1281→    where: { organizationId, resource: "memberships" },
  1282→    orderBy: { updatedAt: "desc" },
  1283→  });
  1284→  const SOURCE_FRESHNESS_MAX_AGE_MS = 24 * 60 * 60 * 1000; // 24 hours
  1285→  const sourceDataFresh = latestCheckpoint
  1286→    ? (now.getTime() - latestCheckpoint.updatedAt.getTime()) < SOURCE_FRESHNESS_MAX_AGE_MS
  1287→    : false;
  1288→
  1289→  // ─── Evaluate each candidate ──────────────────────────────
  1290→
  1291→  // Process in batches to avoid memory pressure
  1292→  const candidateArray = Array.from(qualifyingMemberships);
  1293→
  1294→  for (let batchStart = 0; batchStart < candidateArray.length; batchStart += CANDIDATE_BATCH_SIZE) {
  1295→    const batch = candidateArray.slice(batchStart, batchStart + CANDIDATE_BATCH_SIZE);
  1296→
  1297→    for (const membership of batch) {
  1298→      result.checksPerformed++;
  1299→      const studentId = membership.studentId;
  1300→
  1301→      // Check 9: Membership belongs to mapped product (guaranteed by our query)
  1302→
  1303→      // Check 10: Membership active or trialing (guaranteed by our query)
  1304→
  1305→      // Check 11: Membership not ending (no renewalDate or renewalDate in the future)
  1306→      if (membership.renewalDate && membership.renewalDate <= now) {
  1307→        continue;
  1308→      }
  1309→
  1310→      // Check 12: Activation delay elapsed
  1311→      const daysSinceJoin = (now.getTime() - membership.joinedAt.getTime()) / (1000 * 60 * 60 * 24);
  1312→      if (daysSinceJoin < mapping.activationDelayDays) {
  1313→        continue;
  1314→      }
  1315→
  1316→      // Check 13: Course activity absent
  1317→      const courseState = activityByStudent.get(studentId);
  1318→      if (courseState && courseState.lessonsCompleted > 0) {
  1319→        continue;
  1320→      }
  1321→
  1322→      // Check 14: No course-, campaign- or organisation-level suppression
  1323→      if (suppressedStudentIds.has(studentId)) {
  1324→        continue;
  1325→      }
  1326→
  1327→      // Check 15: No existing equivalent active intervention
  1328→      if (studentsWithActiveIntervention.has(studentId)) {
  1329→        continue;
  1330→      }
  1331→
  1332→      // Campaign cooldown check
  1333→      const lastIvDate = lastInterventionByStudent.get(studentId);
  1334→      if (lastIvDate) {
  1335→        const daysSinceLast = (now.getTime() - lastIvDate.getTime()) / (1000 * 60 * 60 * 24);
  1336→        if (daysSinceLast < campaign.cooldownDays) {
  1337→          continue;
  1338→        }
  1339→      }
  1340→
  1341→      // Check: Per-student message limit
  1342→      const studentRecentCount = recentInterventionCounts.get(studentId) ?? 0;
  1343→      if (studentRecentCount >= campaign.maxMessagesPerStudent) {
  1344→        continue;
  1345→      }
  1346→
  1347→      // Check: Org-wide message limit
  1348→      if (orgWideMessageCount >= campaign.maxMessagesPerOrg) {
  1349→        continue;
  1350→      }
  1351→
  1352→      // Check: Campaign message limit
  1353→      if (campaignMessageCount >= campaign.maxMessagesPerOrg) {
  1354→        continue;
  1355→      }
  1356→
  1357→      // Check 16: Plan allows monitored member
  1358→      if (currentMonitoredMembers >= maxMonitoredMembers) {
  1359→        continue;
  1360→      }
  1361→
  1362→      // Check 17: Source data sufficiently fresh
  1363→      if (!sourceDataFresh) {
  1364→        result.warnings?.push?.("Source data is stale — skipping candidate detection");
  1365→        continue;
  1366→      }
  1367→
  1368→      // Check: Existing snapshot for this window (idempotency)
  1369→      if (studentsWithExistingSnapshot.has(studentId)) {
  1370→        result.snapshotsSkipped++;
  1371→        continue;
  1372→      }
  1373→
  1374→      // ─── Candidate found — create immutable eligibility snapshot ───
  1375→      result.candidatesFound++;
  1376→
  1377→      const evidence = {
  1378→        detectedAt: now.toISOString(),
  1379→        membershipId: membership.id,
  1380→        membershipStatus: membership.status,
  1381→        joinedAt: membership.joinedAt.toISOString(),
  1382→        daysSinceJoin: Math.floor(daysSinceJoin),
  1383→        activationDelayDays: mapping.activationDelayDays,
  1384→        courseId: mapping.courseId,
  1385→        productId: mapping.productId,
  1386→        campaignId,
  1387→        campaignVersionId: latestVersion.id,
  1388→        checks: {
  1389→          orgActive: true,
  1390→          orgNotPaused: true,
  1391→          installationActive: true,
  1392→          campaignActive: true,
  1393→          campaignIsActivationRescue: true,
  1394→          manualApprovalEnabled: true,
  1395→          campaignVersionExists: true,
  1396→          confirmedMappingBelongsToCampaign: true,
  1397→          membershipBelongsToMappedProduct: true,
  1398→          membershipActiveOrTrialing: true,
  1399→          membershipNotEnding: !membership.renewalDate || membership.renewalDate > now,
  1400→          activationDelayElapsed: true,
  1401→          courseActivityAbsent: true,
  1402→          notSuppressed: true,
  1403→          noActiveIntervention: true,
  1404→          campaignCooldownClear: !lastIvDate || (now.getTime() - lastIvDate.getTime()) / (1000 * 60 * 60 * 24) >= campaign.cooldownDays,
  1405→          orgMessageLimitClear: orgWideMessageCount < campaign.maxMessagesPerOrg,
  1406→          campaignMessageLimitClear: campaignMessageCount < campaign.maxMessagesPerOrg,
  1407→          planAllowsMonitoredMember: currentMonitoredMembers < maxMonitoredMembers,
  1408→          sourceDataFresh: true,
  1409→        },
  1410→      };
  1411→
  1412→      // Build idempotency key: student + campaign version + eligibility window
  1413→      const idempotencyKey = `${studentId}:${latestVersion.id}:${eligibilityWindowStart.toISOString()}`;
  1414→
  1415→      try {
  1416→        await db.eligibilitySnapshot.create({
  1417→          data: {
  1418→            organizationId,
  1419→            studentId,
  1420→            campaignId,
  1421→            campaignVersionId: latestVersion.id,
  1422→            idempotencyKey,
  1423→            state: "eligible",
  1424→            evidenceJson: evidence as any,
  1425→            detectedAt: now,
  1426→            eligibilityWindowStart,
  1427→            expiresAt: new Date(now.getTime() + 7 * 24 * 60 * 60 * 1000),
  1428→          },
  1429→        });
  1430→        result.snapshotsCreated++;
  1431→      } catch (error) {
  1432→        const failure = classifyCreateError(error, idempotencyKey);
  1433→        if (
  1434→          failure.class === "duplicate_unique_constraint" ||
  1435→          failure.class === "duplicate_payload_hash"
  1436→        ) {
  1437→          // Idempotent — snapshot already exists for this window
  1438→          result.snapshotsSkipped++;
  1439→        } else {
  1440→          result.errors.push(
  1441→            `Student ${studentId}: ${failure.message}`,
  1442→          );
  1443→        }
  1444→      }
  1445→    }
  1446→  }
  1447→
  1448→  return result;
  1449→}

  1453→/**
  1454→ * Run a full sync with durable execution records and stage tracking.
  1455→ * Returns the sync execution ID for observability.
  1456→ */
  1457→export async function runFullSync(
  1458→  params: SyncExecutionParams & { resources?: string[] },
  1459→): Promise<{ executionId: string; results: SyncResult[] }> {
  1460→  const resources = params.resources ?? ["courses", "products", "memberships"];
  1461→  const results: SyncResult[] = [];
  1462→
  1463→  // Create the execution record
  1464→  const execution = await createSyncExecution({
  1465→    orgId: params.organizationId,
  1466→    trigger: params.requestedBy ? "manual" : "scheduled",
  1467→    requestedBy: params.requestedBy,
  1468→    jobId: params.syncExecutionId,
  1469→  });
  1470→
  1471→  try {
  1472→    for (const resource of resources) {
  1473→      const stage = await createSyncStage(execution.id, resource);
  1474→
  1475→      let syncResult: SyncResult;
  1476→      switch (resource) {
  1477→        case "courses":
  1478→          syncResult = await syncCourses({ ...params, syncExecutionId: execution.id });
  1479→          break;
  1480→        case "products":
  1481→          syncResult = await syncProducts({ ...params, syncExecutionId: execution.id });
  1482→          break;
  1483→        case "memberships":
  1484→          syncResult = await syncMemberships({ ...params, syncExecutionId: execution.id });
  1485→          break;
  1486→        default:
  1487→          await completeSyncStage(stage.id, {
  1488→            pagesProcessed: 0, recordsRead: 0, recordsCreated: 0,
  1489→            recordsUpdated: 0, recordsSkipped: 0,
  1490→          }, "skipped");
  1491→          continue;
  1492→      }
  1493→
  1494→      results.push(syncResult);
  1495→
  1496→      // Complete the stage with counts
  1497→      await completeSyncStage(stage.id, {
  1498→        pagesProcessed: 0, // Pages are tracked inside the sync functions
  1499→        recordsRead: syncResult.recordsRead,
  1500→        recordsCreated: syncResult.recordsCreated,
  1501→        recordsUpdated: syncResult.recordsUpdated,
  1502→        recordsSkipped: syncResult.recordsSkipped,
  1503→      });
  1504→    }
  1505→
  1506→    await completeSyncExecution(execution.id, "completed");
  1507→  } catch (error) {
  1508→    await completeSyncExecution(
  1509→      execution.id,
  1510→      "failed",
  1511→      error instanceof Error ? error.message : "Unknown error",
  1512→    );
  1513→  }
  1514→
  1515→  return { executionId: execution.id, results };
  1516→}

  1520→/**
  1521→ * Phase 10: Bounded aggregation for course state recalculation.
  1522→ *
  1523→ * Instead of N+1 queries (one per student, then 3 more per student),
  1524→ * uses 3 bounded aggregate queries total, then upserts in batches.
  1525→ *
  1526→ * Handles:
  1527→ * - Out-of-order interactions (uses MIN/MAX aggregation)
  1528→ * - Older events arriving after newer events (idempotent upsert)
  1529→ * - Course lesson count changes (recalculates from current course.lessonCount)
  1530→ * - Removed lessons (count only counts existing "completed" events)
  1531→ */
  1532→async function recalculateCourseStatesBounded(organizationId: string, courseId: string): Promise<void> {
  1533→  const course = await db.course.findUnique({ where: { id: courseId } });
  1534→  if (!course) return;
  1535→
  1536→  // Bounded aggregation: group by student in a single query using raw SQL
  1537→  // This replaces the N+1 pattern of: findDistinct students → count per student → findFirst per student
  1538→  const aggregated = await db.$queryRaw<Array<{
  1539→    studentId: string;
  1540→    completedCount: bigint;
  1541→    firstOccurredAt: Date;
  1542→    lastOccurredAt: Date;
  1543→  }>>`
  1544→    SELECT
  1545→      "studentId",
  1546→      COUNT(*) FILTER (WHERE action = 'completed') AS "completedCount",
  1547→      MIN("occurredAt") AS "firstOccurredAt",
  1548→      MAX("occurredAt") AS "lastOccurredAt"
  1549→    FROM "progress_events"
  1550→    WHERE "organizationId" = ${organizationId}
  1551→      AND "courseId" = ${courseId}
  1552→    GROUP BY "studentId"
  1553→  `;
  1554→
  1555→  if (aggregated.length === 0) return;
  1556→
  1557→  // Upsert student course states in batches
  1558→  for (let i = 0; i < aggregated.length; i += PROGRESS_BATCH_SIZE) {
  1559→    const batch = aggregated.slice(i, i + PROGRESS_BATCH_SIZE);
  1560→
  1561→    await Promise.all(
  1562→      batch.map(async (row) => {
  1563→        const completedCount = Number(row.completedCount);
  1564→        const progressPercent = course.lessonCount > 0
  1565→          ? Math.min(100, Math.round((completedCount / course.lessonCount) * 100))
  1566→          : 0;
  1567→
  1568→        await db.studentCourseState.upsert({
  1569→          where: { studentId_courseId: { studentId: row.studentId, courseId } },
  1570→          create: {
  1571→            organizationId,
  1572→            studentId: row.studentId,
  1573→            courseId,
  1574→            progressPercent,
  1575→            lessonsCompleted: completedCount,
  1576→            totalLessons: course.lessonCount,
  1577→            firstActivityAt: row.firstOccurredAt,
  1578→            lastActivityAt: row.lastOccurredAt,
  1579→          },
  1580→          update: {
  1581→            lessonsCompleted: completedCount,
  1582→            progressPercent,
  1583→            totalLessons: course.lessonCount,
  1584→            lastActivityAt: row.lastOccurredAt,
  1585→          },
  1586→        });
  1587→      }),
  1588→    );
  1589→  }
  1590→}
  1591→
  1592→/**
  1593→ * Legacy recalculateCourseStates — kept for backward compatibility.
  1594→ * Delegates to the bounded version.
  1595→ */
  1596→async function recalculateCourseStates(organizationId: string, courseId: string): Promise<void> {
  1597→  return recalculateCourseStatesBounded(organizationId, courseId);
  1598→}
