     1→// Performance and scale verification benchmarks.
     2→//
     3→// Generates repeatable fixture data at production-representative scale and
     4→// measures critical operations against reasonable budgets.
     5→//
     6→// Scale targets:
     7→//   - 100 organisations
     8→//   - 2 500 memberships per org (250 000 total)
     9→//   - Multiple courses per org
    10→//   - Progress events, candidate snapshots, audit events, outbox backlog
    11→//
    12→// Measured operations:
    13→//   1. Course sync page
    14→//   2. Membership sync page
    15→//   3. Candidate batch detection
    16→//   4. Students list
    17→//   5. Queue (intervention) list
    18→//   6. Audit list
    19→//   7. Usage calculation
    20→//   8. Outbox claim
    21→//   9. Reconciliation
    22→//
    23→// Budgets enforced:
    24→//   - No unbounded query (all findMany capped by LIMIT or take)
    25→//   - No list endpoint returns > 100 records per page
    26→//   - No sync page exceeds configured batch size (PAGE_SIZE = 50)
    27→//   - No N+1 growth: query count must NOT be proportional to membership count
    28→//
    29→// IMPORTANT:
    30→//   - We do NOT claim performance is proven without these tests passing.
    31→//   - We do NOT optimise through unsafe caching (no stale reads, no
    32→//     cache-aside that could serve wrong tenant data).
    33→//   - These benchmarks use a real database when DATABASE_URL is set;
    34→//     otherwise they run against simulated query-cost models to verify
    35→//     algorithmic complexity budgets.

    84→/** Simple seeded PRNG for repeatable fixture data. */
    85→function createRng(seed: number) {
    86→  let s = seed | 0;
    87→  return () => {
    88→    s = (s + 0x6d2b79f5) | 0;
    89→    let t = Math.imul(s ^ (s >>> 15), 1 | s);
    90→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    91→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    92→  };
    93→}
    94→
    95→/** Generate a deterministic ID from seed + indices. */
    96→function seededId(prefix: string, orgIdx: number, itemIdx: number): string {
    97→  return `${prefix}_o${orgIdx.toString().padStart(3, "0")}_i${itemIdx.toString().padStart(6, "0")}`;
    98→}

   172→/**
   173→ * Generate repeatable fixture data for the full scale target.
   174→ * Uses a seeded RNG so the dataset is identical across runs.
   175→ */
   176→function generateScaleFixtures(): OrgFixture[] {
   177→  const rng = createRng(0xdeadbeef);
   178→  const orgs: OrgFixture[] = [];
   179→  const baseDate = new Date("2026-01-01T00:00:00Z");
   180→
   181→  for (let o = 0; o < ORG_COUNT; o++) {
   182→    // Courses
   183→    const courses: CourseFixture[] = [];
   184→    for (let c = 0; c < COURSES_PER_ORG; c++) {
   185→      courses.push({
   186→        id: seededId("course", o, c),
   187→        externalId: `ext-${seededId("course", o, c)}`,
   188→        name: COURSE_NAMES[c % COURSE_NAMES.length] ?? `Course ${c}`,
   189→        lessonCount: COURSE_LESSONS[c % COURSE_LESSONS.length] ?? 10,
   190→      });
   191→    }
   192→
   193→    // Product for this org
   194→    const productId = seededId("prod", o, 0);
   195→
   196→    // Memberships
   197→    const memberships: MembershipFixture[] = [];
   198→    for (let m = 0; m < MEMBERSHIPS_PER_ORG; m++) {
   199→      const r = rng();
   200→      let status: MembershipFixture["status"];
   201→      if (r < 0.6) status = "active";
   202→      else if (r < 0.8) status = "trialing";
   203→      else if (r < 0.9) status = "cancelling";
   204→      else status = "cancelled";
   205→
   206→      const joinedDaysAgo = Math.floor(rng() * 120) + 1;
   207→      const joinedAt = new Date(baseDate.getTime() - joinedDaysAgo * 86_400_000);
   208→
   209→      memberships.push({
   210→        id: seededId("mem", o, m),
   211→        userId: seededId("user", o, m),
   212→        productId,
   213→        status,
   214→        joinedAt,
   215→      });
   216→    }
   217→
   218→    // Progress events (~2 per membership on average)
   219→    const progressEvents: ProgressFixture[] = [];
   220→    for (let p = 0; p < PROGRESS_PER_ORG; p++) {
   221→      const mIdx = p % MEMBERSHIPS_PER_ORG;
   222→      const cIdx = p % COURSES_PER_ORG;
   223→      const daysAgo = Math.floor(rng() * 60) + 1;
   224→      progressEvents.push({
   225→        studentId: seededId("user", o, mIdx),
   226→        courseId: courses[cIdx]?.id ?? courses[0].id,
   227→        lessonIdx: Math.floor(rng() * 30),
   228→        completed: rng() > 0.1,
   229→        occurredAt: new Date(baseDate.getTime() - daysAgo * 86_400_000),
   230→      });
   231→    }
   232→
   233→    // Candidate snapshots (40% of memberships are at-risk)
   234→    const candidateSnapshots: CandidateFixture[] = [];
   235→    for (let c = 0; c < CANDIDATES_PER_ORG; c++) {
   236→      const mIdx = c % MEMBERSHIPS_PER_ORG;
   237→      const r = rng();
   238→      let eligibilityState: CandidateFixture["eligibilityState"];
   239→      if (r < 0.7) eligibilityState = "eligible";
   240→      else if (r < 0.9) eligibilityState = "ineligible";
   241→      else eligibilityState = "pending_evaluation";
   242→
   243→      candidateSnapshots.push({
   244→        studentId: seededId("user", o, mIdx),
   245→        courseId: courses[c % COURSES_PER_ORG]?.id ?? courses[0].id,
   246→        eligibilityState,
   247→        snapshotAt: new Date(baseDate.getTime() - Math.floor(rng() * 7) * 86_400_000),
   248→      });
   249→    }
   250→
   251→    // Audit events
   252→    const auditEvents: AuditFixture[] = [];
   253→    const auditActions = ["created", "updated", "approved", "dismissed", "scheduled"];
   254→    const auditObjectTypes = ["intervention", "campaign", "student", "membership", "sync"];
   255→    for (let a = 0; a < AUDIT_EVENTS_PER_ORG; a++) {
   256→      auditEvents.push({
   257→        action: auditActions[Math.floor(rng() * auditActions.length)] ?? "created",
   258→        objectType: auditObjectTypes[Math.floor(rng() * auditObjectTypes.length)] ?? "intervention",
   259→        objectId: seededId("obj", o, a),
   260→        occurredAt: new Date(baseDate.getTime() - Math.floor(rng() * 30) * 86_400_000),
   261→      });
   262→    }
   263→
   264→    // Outbox backlog
   265→    const outboxEvents: OutboxFixture[] = [];
   266→    const outboxTypes = [
   267→      "intervention/created",
   268→      "intervention/scheduled",
   269→      "notification/send",
   270→      "sync/trigger",
   271→    ];
   272→    for (let e = 0; e < OUTBOX_BACKLOG_PER_ORG; e++) {
   273→      const r = rng();
   274→      let state: OutboxFixture["state"];
   275→      if (r < 0.7) state = "pending";
   276→      else if (r < 0.85) state = "dispatching";
   277→      else state = "dispatched";
   278→
   279→      outboxEvents.push({
   280→        eventType: outboxTypes[Math.floor(rng() * outboxTypes.length)] ?? "test/event",
   281→        state,
   282→        createdAt: new Date(baseDate.getTime() - Math.floor(rng() * 7) * 86_400_000),
   283→      });
   284→    }
   285→
   286→    orgs.push({
   287→      orgId: seededId("org", o, 0),
   288→      slug: orgSlug(o),
   289→      name: `Perf Org ${o}`,
   290→      courses,
   291→      memberships,
   292→      progressEvents,
   293→      candidateSnapshots,
   294→      auditEvents,
   295→      outboxEvents,
   296→    });
   297→  }
   298→
   299→  return orgs;
   300→}

   302→// ═══════════════════════════════════════════════════════════════
   303→// Query-cost model (simulated when no DB available)
   304→// ═══════════════════════════════════════════════════════════════
   305→//
   306→// When DATABASE_URL is not set, we simulate query counts based on
   307→// the known implementation of each operation. This lets us verify
   308→// algorithmic complexity budgets without needing a real database.
   309→
   310→interface SimulatedQueryCost {
   311→  queryCount: number;
   312→  rowsProcessed: number;
   313→  rowsReturned: number;
   314→}
   315→
   316→/** Simulate course sync page cost. One page = one findMany + one createMany/updateMany. */
   317→function simulateCourseSyncPage(
   318→  totalCourses: number,
   319→  pageSize: number,
   320→): SimulatedQueryCost {
   321→  const rowsInPage = Math.min(pageSize, totalCourses);
   322→  // loadExternalIds (1 query) + createMany/updateMany (2 queries)
   323→  // + persistCheckpoint (1 upsert)
   324→  return {
   325→    queryCount: 4, // constant regardless of pageSize
   326→    rowsProcessed: rowsInPage,
   327→    rowsReturned: rowsInPage,
   328→  };
   329→}
   330→
   331→/** Simulate membership sync page cost. */
   332→function simulateMembershipSyncPage(
   333→  totalMemberships: number,
   334→  pageSize: number,
   335→): SimulatedQueryCost {
   336→  const rowsInPage = Math.min(pageSize, totalMemberships);
   337→  // loadExternalIds (1) + createMany for new (1) + batch updates (bounded)
   338→  // + persistCheckpoint (1)
   339→  // Worst case all updates: ceil(pageSize / MAX_CONCURRENT_UPDATES)
   340→  return {
   341→    queryCount: 3 + Math.ceil(rowsInPage / MAX_CONCURRENT_UPDATES),
   342→    rowsProcessed: rowsInPage,
   343→    rowsReturned: rowsInPage,
   344→  };
   345→}

   348→function simulateCandidateBatch(

   350→): SimulatedQueryCost {

   352→  return {
   353→    queryCount: 3,
   354→    rowsProcessed: batchSize,
   355→    rowsReturned: batchSize,
   356→  };
   357→}

   456→    },

   460→/** Run a function and measure duration + query count. */
   461→async function measure<T>(
   462→  fn: () => Promise<T>,
   463→): Promise<{ result: T; durationMs: number; queryCount: number }> {
   464→  queryCounter = 0;
   465→  queryCounting = true;
   466→  const start = performance.now();
   467→  const result = await fn();
   468→  const durationMs = performance.now() - start;
   469→  queryCounting = false;
   470→  const qc = queryCounter;
   471→  queryCounter = 0;
   472→  return { result, durationMs, queryCount: qc };
   473→}
