   155→  it("handles duplicate interactions by skipping (not erroring)", () => {
   156→    // Simulate: same external interaction ID arrives twice
   157→    // The unique constraint [organizationId, studentId, externalInteractionId]
   158→    // prevents double-counting. skipDuplicates silently skips.
   159→    const seen = new Set<string>();
   160→    const interactionId = "int-123";

   162→    // First arrival

   166→    // Second arrival (duplicate) — check before create

   173→  it("handles out-of-order interactions by using source timestamp", () => {
   174→    // Events may arrive out of chronological order.
   175→    // The payloadHash includes sourceTimestamp, so the same event
   176→    // at a different source time produces a different hash.
   177→    // The recalculation uses MIN/MAX aggregation which is order-independent.
   178→    const events = [
   179→      { sourceTimestamp: "2026-01-03T00:00:00Z", action: "completed" }, // Arrives first
   180→      { sourceTimestamp: "2026-01-01T00:00:00Z", action: "started" },   // Arrives second
   181→    ];
   182→
   183→    // After bounded aggregation, firstActivityAt = MIN(occurredAt) = Jan 1
   184→    // lastActivityAt = MAX(occurredAt) = Jan 3
   185→    const timestamps = events.map((e) => new Date(e.sourceTimestamp).getTime());
   186→    const firstActivityAt = new Date(Math.min(...timestamps));
   187→    const lastActivityAt = new Date(Math.max(...timestamps));
   188→
   189→    expect(firstActivityAt.toISOString()).toBe("2026-01-01T00:00:00.000Z");
   190→    expect(lastActivityAt.toISOString()).toBe("2026-01-03T00:00:00.000Z");
   191→  });

   193→  it("handles older events arriving after newer events", () => {
   194→    // An old "started" event arrives after a newer "completed" event.
   195→    // The payload hash is different (different action + timestamp),
   196→    // so both are stored. Recalculation counts "completed" actions only.
   197→    const events = [
   198→      { id: "int-1", action: "completed", sourceTimestamp: "2026-01-02" },

   250→  it("handles students with activity but no qualifying membership", () => {
   251→    // A student has course interactions but no membership for the mapped product.

   253→    // and flagged as "course_activity_without_membership" during reconciliation.
   254→    const studentId = "student-123";
   255→    const hasMembership = false;
   256→    const hasActivity = true;
   257→
   258→    // Progress ingestion: skip with warning
   259→    let skipped = false;
   260→    let warning = "";
   261→    if (!hasMembership) {
   262→      skipped = true;
   263→      warning = `Student ${studentId} not found — skipped (membership may arrive later)`;
   264→    }

   266→    expect(warning).toContain("membership may arrive later");

   268→    // Reconciliation: classified as course_activity_without_membership

   270→    if (hasActivity && !hasMembership) {

   276→  it("handles membership arriving after course activity", () => {
   277→    // Course activity was synced first, membership synced later.
   278→    // On next reconciliation, the student moves from
   279→    // "course_activity_without_membership" to "matched".
   280→    const stateBeforeMembership = { hasMembership: false, hasActivity: true };
   281→    const stateAfterMembership = { hasMembership: true, hasActivity: true };
   282→
   283→    function classify(s: { hasMembership: boolean; hasActivity: boolean }) {
   284→      if (s.hasMembership && s.hasActivity) return "matched";
   285→      if (s.hasMembership && !s.hasActivity) return "membership_without_course_activity";
   286→      if (!s.hasMembership && s.hasActivity) return "course_activity_without_membership";
   287→      return "unmapped";
   288→    }
   289→
   290→    expect(classify(stateBeforeMembership)).toBe("course_activity_without_membership");
   291→    expect(classify(stateAfterMembership)).toBe("matched");
   292→  });

   667→describe("result accounting (cross-phase)", () => {
   668→  it("tracks create vs update counts correctly", () => {
   669→    const result = {
   670→      recordsRead: 0,
   671→      recordsCreated: 0,
   672→      recordsUpdated: 0,
   673→      recordsSkipped: 0,
   674→    };
   675→
   676→    const items = [
   677→      { exists: true },   // update
   678→      { exists: false },  // create
   679→      { exists: false },  // create
   680→      { exists: true },   // update
   681→      { exists: false },  // create
   682→      { exists: true },   // update
   683→      { exists: false },  // create
   684→      { exists: false },  // create
   685→      { error: true },    // skip
   686→      { error: true },    // skip
   687→    ];
   688→
   689→    for (const item of items) {
   690→      result.recordsRead++;
   691→      if (item.error) {
   692→        result.recordsSkipped++;
   693→      } else if ((item as any).exists) {
   694→        result.recordsUpdated++;
   695→      } else {
   696→        result.recordsCreated++;
   697→      }
   698→    }
   699→
   700→    expect(result.recordsRead).toBe(10);
   701→    expect(result.recordsCreated).toBe(5);
   702→    expect(result.recordsUpdated).toBe(3);
   703→    expect(result.recordsSkipped).toBe(2);
   704→    expect(result.recordsRead).toBe(
   705→      result.recordsCreated + result.recordsUpdated + result.recordsSkipped,
   706→    );
   707→  });
